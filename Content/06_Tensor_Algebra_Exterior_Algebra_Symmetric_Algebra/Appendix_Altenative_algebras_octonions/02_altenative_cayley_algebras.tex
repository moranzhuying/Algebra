\section{交错Cayley代数}

记号约定:$A$是交换环.

\begin{proposition}{交错Cayley代数的性质}{}
	设$F$是$A$上的Caylay代数,单位元是$e$,共轭映射是$s:F\to F,x\mapsto\bar{x}$,Caylay范数为$\mathrm{N}:F\to A$.
	\begin{enumerate}
		\item $F$是交错代数$\iff$ $\forall x,y\in F\left(x^{2}y=x\left(xy\right)\right)$.
		\item 若$F$是交错代数,则$\forall x,y\in F\left(\mathrm{N}\left(xy\right)=\mathrm{N}\left(x\right)\mathrm{N}\left(y\right)\right)$.
		\item 设$F$是交错代数,$x\in F$,则$x\in F^{\times}\iff\mathrm{N}\left(x\right)\in\left(Ae\right)^{\times}$.若$x\in F^{\times}$,则$x^{-1}$唯一并等于$\left(\mathrm{N}\left(x\right)\right)^{-1}\bar{x}$,同时
		\begin{align*}
			\forall y\in F\left(x^{-1}\left(xy\right)=x\left(x^{-1}y\right)=y\right).
		\end{align*}
	\end{enumerate}
\end{proposition}

\begin{proposition}{Cayley扩张的交错性判断}{}
	设$E$是$A$上的Cayley代数,$\gamma\in A$,$F$是由$\gamma$和$s$定义的Caylay扩张,则$F$是交错代数$\iff$ $E$是结合代数.
\end{proposition}
