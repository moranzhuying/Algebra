\section{有限生成自由模的左缩并和右缩并}

记号约定:$M$是$n$维有限生成自由$A$-模,$\left(e_{i}\right)_{i=1}^{n}$是$M$的基,$\left(e_{i}^{\ast}\right)_{i=1}^{n}$是其对偶基.

\begin{proposition}{}{}
	对每个有限序列$s\coloneq\left(s_{l}\right)_{l=1}^{p},t\coloneq\left(t_{s}\right)_{s=1}^{q}$,有$e_{s}\llcorner e_{t}=
	\begin{cases}
		e_{u},&\text{存在序列}u\coloneq\left(u_{k}\right)_{k=1}^{p+q}\text{使得}u=st,\\
		0,&\text{不存在序列}u\coloneq\left(u_{k}\right)_{k=1}^{p+q}\text{使得}u=st.
	\end{cases}$
\end{proposition}
