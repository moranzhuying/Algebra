\section{与\texorpdfstring{$p$}{p}-向量关联的子空间的应用}

记号约定:$K$是除环,$E$是$K$-向量空间.

\begin{definition}{$p$-关联子空间}{}
	对每个$z\in\bigwedge\limits^{p}\left(E\right)$,定义$z$的$p$-关联子空间$M_{z}$是$E$中满足$z\in\bigwedge\limits^{p}\left(M\right)$的最小有限维子空间$M$.
\end{definition}

\begin{proposition}{}{}
	设$z\in\bigwedge\limits^{p}\left(E\right)$.定义$\lambda_{z}:\bigwedge\limits^{p-1}\left(E^{\vee}\right)\to E,u^{\ast}\mapsto u^{\ast}\lrcorner z$,则$M_{z}^{\perp}=\ker\left(\lambda_{z}\right),M_{z}=\im\left(\lambda_{z}\right)$.
\end{proposition}
