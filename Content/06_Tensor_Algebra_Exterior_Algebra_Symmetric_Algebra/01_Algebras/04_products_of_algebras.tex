\section{代数的乘积}

记号约定:$A$是交换环,$\left(E_{i}\right)_{i\in I}$是一族$A$-代数,$E=\prod\limits_{i\in I}E_{i}$.

\begin{definition}{代数的乘积}{}
	定义$E$上的运算$\cdot$为$\left(\left(x_{i}\right)_{i\in I},\left(y_{i}\right)_{i\in I}\right)\mapsto\left(x_{i}y_{i}\right)_{i\in I}$,则$E$是$A$-代数,称为$\left(E_{i}\right)_{i\in I}$的乘积.
\end{definition}

\begin{proposition}{代数的乘积继承交换性,结合性和幺元}{}
	若$\left(E_{i}\right)_{i\in I}$是一族$A$-结合(相对的,交换,含幺)代数,则$E$是$A$-结合(相对的,交换,含幺)代数.
\end{proposition}

\begin{remark}{环的乘积的性质在代数的推广}{}
	关于环的乘积的所有性质,不需要做任何修改就可以应用到代数的乘积上.
\end{remark}
