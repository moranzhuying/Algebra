\section{代数的基,乘法表}

记号约定:$A$是交换环,$E$是$A$-代数.

\begin{definition}{代数的基}{}
	我们称$E$作为$A$-模的基是$E$(作为代数)的基.
\end{definition}

\begin{proposition}{$A$-代数唯一确定结构常数}{}
	设$\left(e_{i}\right)_{i\in I}$是$E$的基,则存在$A$的元素族$\left(\gamma_{i,j}^{k}\right)_{i,j,k\in I}$满足如下条件:
	\begin{itemize}
		\item $\left(\left(\gamma_{i,j}^{k}\right)_{k\in I}\right)_{i,j\in I}$是一族有限支撑族.
		\item 对任意$i,j\in I$有$e_{i}e_{j}=\sum\limits_{k\in I}\gamma_{i,j}^{k}e_{k}$.
	\end{itemize}
\end{proposition}

\begin{proposition}{结构常数唯一确定$A$-代数}{}
	设$M$是自由$A$-模,其基为$\left(e_{i}\right)_{i\in I}$,$A$中的元素族$\left(\gamma_{i,j}^{k}\right)$满足如下条件:
	\begin{center}
		$\left(\left(\gamma_{i,j}^{k}\right)_{k\in I}\right)_{i,j\in I}$是一族有限支撑族.
	\end{center}
	则$M$上存在唯一的运算使得$M$是$A$-代数,并且对任意$i,j\in I$有$e_{i}e_{j}=\sum\limits_{k\in I}\gamma_{i,j}^{k}e_{k}$.
\end{proposition}


\begin{definition}{代数的结构常数,乘法表}{}
	设$\left(e_{i}\right)_{i\in I}$是$E$的基,$A$的元素族$\left(\gamma_{i,j}^{k}\right)_{i,j,k\in I}$满足如下条件:
	\begin{itemize}
		\item $\left(\left(\gamma_{i,j}^{k}\right)_{k\in I}\right)_{i,j\in I}$是一族有限支撑族.
		\item 对任意$i,j\in I$有$e_{i}e_{j}=\sum\limits_{k\in I}\gamma_{i,j}^{k}e_{k}$.
	\end{itemize}
	我们称$\left(\gamma_{i,j}^{k}\right)_{i,j,k\in I}$是$E$关于$\left(e_{i}\right)_{i\in I}$的结构常数,称关系式$\forall i,j\in I\left(e_{i}e_{j}=\sum\limits_{k\in I}\gamma_{i,j}^{k}e_{k}\right)$为$E$的乘法表.
\end{definition}

\begin{proposition}{生成元确定代数的性质}{}
	设$\left(e_{i}\right)_{i\in I}$是$E$作为$A$-模的生成系.
	\begin{enumerate}
		\item $E$是结合代数$\iff$ $\forall i,j,k\in I\left(\left(e_{i}e_{j}\right)e_{k}=e_{i}\left(e_{j}e_{k}\right)\right)$.
		\item $E$是交换代数$\iff$ $\forall i,j\in I\left(e_{i}e_{j}=e_{j}e_{i}\right)$.
		\item 设$c\in E$,则$c$是$E$的单位元$\iff$ $\forall i\in I\left(ce_{i}=e_{i}c=e_{i}\right)$.
	\end{enumerate}
\end{proposition}

\begin{proposition}{结合律和交换律的结构常数刻画}{}
	设$\left(e_{i}\right)_{i\in I}$是$E$的基,相应的结构常数为$\left(\gamma_{i,j}^{k}\right)_{i,j,k\in I}$.
	\begin{enumerate}
		\item $E$是结合代数$\iff$ $\forall i,j,k,s\in I\left(\sum\limits_{r\in I}\gamma_{i,j}^{r}\gamma_{r,k}^{s}=\sum\limits_{r\in I}\gamma_{i,r}^{s}\gamma_{j,k}^{r}\right)$.
		\item $E$是交换代数$\iff$ $\forall i,j,k\in I\left(\gamma_{i,j}^{k}=\gamma_{j,i}^{k}\right)$.
	\end{enumerate}
\end{proposition}

\begin{proposition}{标量扩张下的结构常数}{}
	设$E$的基是$\left(e_{i}\right)_{i\in I}$,相应的结构常数为$\left(\gamma_{i,j}^{k}\right)_{i,j,k\in I}$,$B$是交换环,$\rho\in\Hom_{\mathsf{Ring}}\left(A,B\right)$,则
	\begin{enumerate}
		\item $\left(\rho^{\ast}\left(e_{i}\right)\right)_{i\in I}$是$\rho^{\ast}\left(E\right)$的基.
		\item $\rho^{\ast}\left(E\right)$关于$\left(\rho^{\ast}\left(e_{i}\right)\right)_{i\in I}$的结构常数是$\left(\rho\left(\gamma_{i,j}^{k}\right)\right)_{i,j,k\in I}$.
	\end{enumerate}
\end{proposition}
