\section{标量的限制与延拓}

\begin{definition}{代数的标量限制}{}
	设$A_{0},A$是交换环,$\rho\in\Hom_{\mathsf{Ring}}\left(A_{0},A\right)$,$E$是$A$-代数,则$\rho_{\ast}\left(E\right)$是$A_{0}$-代数,称之为$E$通过把标量从$A$限制到$A_{0}$得到.
\end{definition}

\begin{definition}{半代数同态}{}
	设$A_{0},A$是交换环,$\rho\in\Hom_{\mathsf{Ring}}\left(A_{0},A\right)$,$E,F$分别是$A$-代数,$A_{0}$-代数.$\Hom_{A-\mathsf{Alg}}\left(F,\rho^{\ast}\left(E\right)\right)$的元素叫$\rho$-半代数同态.
\end{definition}

\begin{proposition}{同态的标量限制}{}
	设$A_{0},A$是交换环,$E,E^{\prime}$分别是$A$-代数,$A_{0}$-代数,$\rho\in\Hom_{\mathsf{Ring}}\left(A_{0},A\right)$.若$f\in\Hom_{A-\mathsf{Alg}}\left(E,E^{\prime}\right)$,则
	\begin{align*}
		\rho_{\ast}\left(f\right)\in\Hom_{A_{0}-\mathsf{Alg}}\left(\rho_{\ast}\left(E\right),\rho_{\ast}\left(E^{\prime}\right)\right).
	\end{align*}
\end{proposition}

\begin{definition}{代数的标量延拓}{}
	设$A,B$是交换环,$\rho\in\Hom_{\mathsf{Ring}}\left(A,B\right)$,$E$是$A$-代数,则$\rho^{\ast}\left(E\right)$是典范的$B$-代数,称为$E$通过$\rho$从$A$延拓到$B$得到的代数.
\end{definition}

\begin{proposition}{标量延拓保持结合性,交换性和幺元}{}
	设$A,B$是交换环,$\rho\in\Hom_{\mathsf{Ring}}\left(A,B\right)$,$E$是$A$-代数.若$E$是结合(相对的,交换,含幺)代数,则$\rho^{\ast}\left(E\right)$亦然.
\end{proposition}

\begin{proposition}{代数的标量延拓的泛性质}{}
	设$A,B$是交换环,$\rho\in\Hom_{\mathsf{Ring}}\left(A,B\right)$,$E$是$A$-代数.
	\begin{enumerate}
		\item 典范映射$\phi_{E}:E\to\rho^{\ast}\left(E\right),a\mapsto 1\otimes a$是$A$-代数同态.
		\item 对任意$B$-代数$F$和$f\in\Hom_{A-\mathsf{Alg}}\left(E,\rho_{\ast}\left(F\right)\right)$,存在唯一的$\bar{f}\in\Hom_{B-\mathsf{Alg}}\left(\rho_{\ast}\left(E\right),F\right)$使下图交换.
		\begin{center}
			\begin{tikzcd}
				E && {\rho_{\ast}\left(\rho^{\ast}\left(E\right)\right)} \\
				& {\rho_{\ast}\left(F\right)}
				\arrow["{\phi_{E}}", from=1-1, to=1-3]
				\arrow["f"', from=1-1, to=2-2]
				\arrow["{\exists!\rho_{\ast}\left(\bar{f}\right)}", dashed, from=1-3, to=2-2]
			\end{tikzcd}
		\end{center}
		\item 存在典范双射$\Hom_{A-\mathsf{Alg}}\left(E,\rho_{\ast}\left(F\right)\right)\to\Hom_{B-\mathsf{Alg}}\left(\rho^{\ast}\left(A\right),B\right)$.
	\end{enumerate}
\end{proposition}

\begin{corollary}{代数的标量扩张的函子性}{}
	设$A,B$是交换环,$\rho\in\Hom_{\mathsf{Ring}}\left(A,B\right)$,$E,F$是$A$-代数.对任意$f\in\Hom_{A-\mathsf{Alg}}\left(E,F\right)$,以下图表交换.
	\begin{center}
		\begin{tikzcd}
			E && {\rho^{\ast}\left(E\right)} \\
			\\
			F && {\rho^{\ast}\left(F\right)}
			\arrow["{\phi_{E}}", from=1-1, to=1-3]
			\arrow["f"', from=1-1, to=3-1]
			\arrow["{f\otimes\id_{B}}", from=1-3, to=3-3]
			\arrow["{\phi_{F}}"', from=3-1, to=3-3]
		\end{tikzcd}
	\end{center}
\end{corollary}

\begin{proposition}{代数的标量扩张的传递同构}{}
	设$A,B,C$是交换环,$\rho\in\Hom_{\mathsf{Ring}}\left(A,B\right),\sigma\in\Hom_{\mathsf{Ring}}\left(B,C\right)$,$E$是$A$-代数,则有典范$C$-代数同构
	\begin{align*}
		\sigma^{\ast}\left(\rho^{\ast}\left(E\right)\right)\simeq\left(\sigma\circ\rho\right)^{\ast}\left(E\right).
	\end{align*}
\end{proposition}
