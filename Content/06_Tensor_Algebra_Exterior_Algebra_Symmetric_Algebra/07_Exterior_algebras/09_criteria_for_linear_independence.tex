\section{线性无关的判别准则}

记号约定:$A$是交换环,$M$是$A$-模.

\begin{proposition}{}{}
	设$M$是投射模,$\left(x_{i}\right)_{i=1}^{n}$是$M$中的有限序列,则$\left(x_{i}\right)_{i=1}^{n}$是自由族$\iff$存在$\lambda\in A\setminus\left\{0\right\}\left(\lambda x_{1}\wedge\ldots\wedge x_{n}=0\right)$.
\end{proposition}

\begin{corollary}{}{}
	设$M,N$是投射$A$-模,$f\in\Hom_{A}\left(M,N\right)$.如果$f$是单射,那么$\bigwedge\left(f\right)$是单射.
\end{corollary}

\begin{proposition}{}{}
	设$N$是$M$的$p$维自由子模,并且是$M$的直因子,$u$是$\bigwedge\limits^{p}\left(N\right)$的生成元,则$\forall x\in M\left(x\in N\iff u\wedge x=0\right)$.
\end{proposition}

\begin{theorem}{}{}
	设$M$是自由模,它的基是$\left(e_{i}\right)_{i=1}^{n}$,而$\left(x_{i}\right)_{i=1}^{n}$是$M$中的序列,则
	\begin{center}
		$\left(x_{i}\right)_{i=1}^{n}$是$M$的基$\iff$ $\forall\lambda\in A\left(x_{1}\wedge\ldots\wedge x_{n}=\lambda\left(e_{1}\wedge\ldots\wedge e_{n}\right)\implies\lambda\in A^{\times}\right)$.
	\end{center}
\end{theorem}
