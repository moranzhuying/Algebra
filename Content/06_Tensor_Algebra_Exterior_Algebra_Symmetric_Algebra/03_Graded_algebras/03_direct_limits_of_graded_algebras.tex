\section{分次代数的正向极限}

记号约定:$I$是预序集,$\left(\left(A_{\alpha}\right)_{\alpha\in I},\left(\phi_{\beta\alpha}\right)_{\alpha,\beta\in I}\right)$是$\Delta$型分次交换环的正系统.

\begin{definition}{$\Delta$型分次代数的正系统}{}
	设$\left(\left(E_{\alpha}\right)_{\alpha\in I},\left(f_{\beta\alpha}\right)_{\alpha,\beta\in I}\right)$是$A_{\alpha}$-模的正系统.若
	\begin{enumerate}
		\item 对每个$\alpha\in I$,$E_{\alpha}$是$\Delta$型分次$A_{\alpha}$-模,
		\item 对每个$\alpha,\beta\in I$且$\alpha\leq\beta$,$f_{\beta\alpha}$是$\Delta$型分次$A_{\alpha}$-代数同态,
	\end{enumerate}
	则称$\left(\left(E_{\alpha}\right)_{\alpha\in I},\left(f_{\beta\alpha}\right)_{\alpha,\beta\in I}\right)$是$\left(\left(A_{\alpha}\right)_{\alpha\in I},\left(\phi_{\beta\alpha}\right)_{\alpha,\beta\in I}\right)$上的$\Delta$型分次代数的正系统.
	
	此时,$\varinjlim E_{\alpha}$是典范的$\Delta$型分次$\varinjlim A_{\alpha}$-代数,称之为$\left(\left(E_{\alpha}\right)_{\alpha\in I},\left(f_{\beta\alpha}\right)_{\alpha,\beta\in I}\right)$是$\left(\left(A_{\alpha}\right)_{\alpha\in I},\left(\phi_{\beta\alpha}\right)_{\alpha,\beta\in I}\right)$的正向极限.
\end{definition}

\begin{theorem}{$\Delta$型分次代数的正向极限的泛性质}{}
	$\left(\left(E_{\alpha}\right)_{\alpha\in I},\left(f_{\beta\alpha}\right)_{\alpha,\beta\in I}\right)$是$\left(\left(A_{\alpha}\right)_{\alpha\in I},\left(\phi_{\beta\alpha}\right)_{\alpha,\beta\in I}\right)$上的$\Delta$型分次代数的正系统.
	\begin{enumerate}
		\item 对每个$\alpha\in I$,典范映射$f_{\alpha}:E_{\alpha}\to\varinjlim E_{\alpha}$是$\Delta$型分次$A_{\alpha}$-代数同态.
		\item 设$F$是$\Delta$型分次代数,对每个$\alpha\in I$,$u_{\alpha}:E_{\alpha}\to F$是$\Delta$型$A_{\alpha}$-代数同态,则$\varinjlim u_{\alpha}$是$\Delta$型分次$\varinjlim A_{\alpha}$-代数同态.
	\end{enumerate}
\end{theorem}
