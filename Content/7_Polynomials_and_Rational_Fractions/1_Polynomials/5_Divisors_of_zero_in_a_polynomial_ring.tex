\section{多项式环中的零因子}

\begin{proposition}{首项系数为可消去元的多项式次数公式}{}
	设$f\in A[X]$非零,其首项系数为$\alpha$.若$\alpha$可消去,则对$A[X]$的每个非零元$g$有$fg\neq 0$和$\deg\left(fg\right)=\deg\left(f\right)+\deg\left(g\right)$.
\end{proposition}

\begin{proposition}{整环上的多项式环是整环}{}
	若$A$是整环,则$A[(X_{i})_{i\in I}]$是整环.
\end{proposition}

\begin{corollary}{整环上的多项式次数公式}{}
	设$A$是整环,则对每个$u,v\in A[(X_{i})_{i\in I}]$有$\deg\left(uv\right)=\deg\left(u\right)+\deg\left(v\right)$.
\end{corollary}

\begin{corollary}{整环上的多项式环中的可逆元}{}
	设$A$是整环,则$\left(A[(X_{i})_{i\in I}]\right)^{\times}=\left\{\alpha X_{i}^{0}\middle|i\in I,\alpha\in A^{\times}\right\}$.
\end{corollary}

\begin{proposition}{多项式环中的幂零元}{}
	设$u\in A[(X_{i})_{i\in I}]$,则$u$幂零$\iff$ $u$的各项系数在$A$中都幂零.
\end{proposition}

\begin{remark}{}{}
	设$A$是整环,$u,v\in A[(X_{i})_{i\in I}]$.若$v$是$u$的非零倍式,$v$是齐次多项式,则$u$是齐次多项式.
\end{remark}
