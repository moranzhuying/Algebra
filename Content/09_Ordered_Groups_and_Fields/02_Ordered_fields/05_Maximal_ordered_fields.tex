\section{极大有序域}

记号约定:$\left(K,+,\cdot,\leq\right)$是有序域,

\begin{definition}{极大有序域}{}
	如果$K$的每个有序代数扩张都是平凡的,则称$K$是极大有序域.
\end{definition}

\begin{theorem}{极大有序代数扩张的存在性}{}
	每个有序域都有一个有序代数扩张是极大有序域.
\end{theorem}

\begin{proposition}{极大有序域上多项式的介值星}{}
	设$K$是极大有序域,$a,b\in K$且$a<b$,$f\in K\left[X\right]$.若$f\left(a\right)f\left(b\right)<0$,则$\exists a<\xi<b\left(f\left(\xi\right)=0\right)$.
\end{proposition}

\begin{proposition}{极大有序域中平方根与奇数次根的存在性}{}
	设$K$是极大有序域,则
	\begin{enumerate}
		\item $K_{>0}$的每个元素在$K$中有平方根,
		\item $K\left[X\right]$的每个奇数次多项式在$K$中有一个根.
	\end{enumerate}
\end{proposition}

\begin{corollary}{极大有序域上全序的刚性}{}
	若$F$为极大有序域,则$F$上存在唯一的全序$\preccurlyeq$使得$\left(F,+,\cdot,\preccurlyeq\right)$为全序域.
\end{corollary}
