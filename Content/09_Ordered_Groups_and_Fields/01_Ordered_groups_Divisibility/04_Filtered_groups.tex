\section{定向群}

记号约定:$\left(G,+,\leq\right)$是有序群.

\begin{proposition}{定向群的等价定义}{equivalent-condition-of-filtered-group}
	$G$按$\leqslant$是右定向集$\iff$ $G$按$\leqslant$是左定向集$\iff$ $G$由$G_{\geq 0}$生成.
\end{proposition}

\begin{definition}{定向群}{}
	若$G$满足命题(\ref{prop:equivalent-condition-of-filtered-group})的三个条件之一,则称$G$是定向群.
\end{definition}

\begin{proposition}{有限元素族的平移}{}
	设$\left(x_{i}\right)_{i\in I}$是$G$中的有限元素族,则$\exists z\in G\forall i\in I\left(x_{i}+z\in G_{\geq 0}\right)$.
\end{proposition}
