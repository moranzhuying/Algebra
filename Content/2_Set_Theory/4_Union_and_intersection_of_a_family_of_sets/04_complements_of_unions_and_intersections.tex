\section{交与并的补集}

\begin{proposition}{子集族的交和并的De Morgan律}{}
	设$\left(X_{i}\right)_{i\in I}$是集合$E$的子集族,则$E\setminus\left(\bigcup\limits_{i\in I}X_{i}\right)=\bigcap\limits_{i\in I}\left(E\setminus X_{i}\right),E\setminus\left(\bigcap\limits_{i\in I}X_{i}\right)=\bigcup\limits_{i\in I}\left(E\setminus X_{i}\right)$.
\end{proposition}
