\section{部分乘积}

\begin{definition}{集合族的乘积的部分乘积}{}
	设$\left(X_{i}\right)_{i\in I}$是集合族,$J\subseteq I$.我们称$\prod\limits_{i\in J}X_{i}$是$\prod\limits_{i\in I}X_{i}$的部分乘积,称$\pr_{J}:\prod\limits_{i\in I}X_{i}\to\prod\limits_{i\in J}X_{i},f\mapsto f|_{J}$为关于$J$的部分投影.
\end{definition}

\begin{proposition}{}{}
	设$\left(X_{i}\right)_{i\in I}$是集合族,$J\subseteq I$.若对任一$i\in I$有$X_{i}\neq\emptyset$,则$\pr_{J}$是满射.
\end{proposition}

\begin{proposition}{部分乘积上的映射的延拓}{}
	设$\left(X_{i}\right)_{i\in I}$是一族非空集合,$J\subseteq I$.若$g\in\prod\limits_{i\in J}X_{i}$,则存在$f\in\prod\limits_{i\in I}X_{i}$使得$f|_{J}=g$.
\end{proposition}

\begin{corollary}{}{}
	设$\left(X_{i}\right)_{i\in I}$是一族非空集合,则$\left(\pr_{i}\right)_{i\in I}$是一族满射.
\end{corollary}

\begin{corollary}{选择公理的集合乘积形式}{}
	设$\left(X_{i}\right)_{i\in I}$是集合族,则$\prod\limits_{i\in I}X_{i}=\emptyset$ $\iff$存在$i\in I$使得$X_{i}=\emptyset$.
\end{corollary}

\begin{corollary}{}{}
	设$\left(X_{i}\right)_{i\in I},\left(Y_{i}\right)_{i\in I}$是集合族.
	\begin{enumerate}
		\item 若对任意$i\in I$有$X_{i}\subseteq Y_{i}$,则$\prod\limits_{i\in I}X_{i}\subseteq\prod\limits_{i\in I}Y_{i}$.
		\item 若$\left(X_{i}\right)_{i\in I}$是一族非空集合,$\prod\limits_{i\in I}X_{i}\subseteq\prod\limits_{i\in I}Y_{i}$,则对任意$i\in I$有$X_{i}\subseteq Y_{i}$.
	\end{enumerate}
\end{corollary}
