\section{自由模上的对称张量}

\begin{proposition}{自由模的对称张量代数的基}{}
	设$M$是自由模,其基为$\left(e_{i}\right)_{i\in I}$.
	\begin{enumerate}
		\item 对每个$\nu\coloneq\left(\nu_{i}\right)_{i\in I}\in\N^{\left(I\right)}$,记$e_{\nu}=\prod\limits_{i\in I}\gamma_{\nu_{i}}\left(e_{i}\right)$,则$\left(e_{\nu}\right)_{\nu\in\N^{\left(I\right)}}$是$\mathsf{TS}\left(M\right)$的基.特别地,$\left(\gamma_{k}\left(x\right)\right)_{k\in\N,x\in M}$生成$\mathsf{TS}\left(M\right)$.
		\item 对每个$p\in\N$,在等同意义下$\mathsf{TS}^{p}\left(M\right)$是$\mathsf{T}^{p}\left(M\right)$的直因子.
	\end{enumerate}
\end{proposition}

\begin{lemma}{有限群作用下的不变子模与补子模}{}
	设$G$有限,$U$是左$A[G]$-模,并且有一个$G$-不变基集$B$.令$\Omega=B/G$,对每个$\omega\in\Omega$,记$u_{\omega}=\sum\limits_{b\in\omega}b$.
	\begin{enumerate}
		\item $\left(u_{\omega}\right)_{\omega\in\Omega}$是$G$-不变子模$U^{G}$的基.
		\item 对每个$\omega\in\Omega$,设$v_{\omega}\in\Omega,\omega^{\prime}=\omega\setminus\left\{v_{\omega}\right\}$.令$B^{\prime}=\bigcup\limits_{\omega\in\Omega}\omega^{\prime}$,则$B^{\prime}$是$U^{G}$在$U$中的补子模.
	\end{enumerate}
\end{lemma}

\begin{proposition}{无限整环上分划幂的生成性}{}
	设$M$是自由模,$k\in\N$.若$A$是无限整环,则$\forall z\in\mathsf{TS}^{k}\left(M\right)\exists\alpha\in A\setminus\left\{0\right\}\left(\alpha z\in\gamma_{k}\left(M\right)\right)$.
\end{proposition}
