\section{对称张量的定义}

记号约定:从本节开始,我们默认$M,N,P$是$A$-模.

\begin{definition}{$n$阶对称张量模}{}
	对每个$n\in\N$,令$\mathsf{TS}^{n}\left(M\right)=\left\{z\in\mathsf{T}^{n}\left(M\right)\middle|\forall\sigma\in\mathfrak{S}_{n}\left(\sigma z=z\right)\right\}$.
\end{definition}

\begin{definition}{对称张量模}{}
	我们称$\mathsf{TS}\left(M\right)\coloneq\bigboxplus\limits_{n\in\N}\mathsf{TS}^{n}\left(M\right)$是$M$的对称张量模.
\end{definition}

\begin{remark}{}{}
	把$\left(\mathsf{TS}^{n}\left(M\right)\right)_{n\in\N}$的各项都等同于$\mathsf{TS}\left(M\right)$的子模,则$\mathsf{TS}\left(M\right)$是$\mathsf{T}\left(M\right)$的分次$A$-子模.
\end{remark}
