\section{对称有理分式}

记号约定:$K$是域,$\left(X_{i}\right)_{i=1}^{n}$是一族不定元.

\begin{proposition}{}{}
	\begin{enumerate}
		\item 对每个$\sigma\in\mathfrak{S}_{n}$,我们总可以把$\varphi_{\sigma}\in\Aut_{A-\mathsf{Alg}}\left(K[X_{1},\ldots,X_{n}]\right)$唯一地延拓为$\psi_{\sigma}\in\Aut_{A-\mathsf{Alg}}\left(K(X_{1},\ldots,X_{n})\right)$.
		\item $\mathfrak{S}_{n}\to K[X_{1},\ldots,X_{n}],\sigma\mapsto\psi_{\sigma}$既是群同态又是单射.
	\end{enumerate}
\end{proposition}

\begin{definition}{对称有理分式}{}
	设$f\in K\left(X_{1},\ldots,X_{n}\right)$.若对每个$\sigma\in\mathfrak{S}_{n}$有$\psi_{\sigma}\left(f\right)=f$,则称$f$是对称有理分式.
\end{definition}

\begin{proposition}{对称有理分式域的代数结构}{}
	$K(X_{1},\ldots,X_{n})$中的对称有理分式组成的集合是$\Frac\left(K[X_{1},\ldots,X_{n}]^{\mathfrak{S}_{n}}\right)$,它是$K(X_{1},\ldots,X_{n})$的子域.
\end{proposition}

\begin{corollary}{对称有理分式域的同构生成订立}{}
	设$e_{1},\ldots,e_{n}\in K[X_{1},\ldots,X_{n}]$是初等对称多项式.
	\begin{enumerate}
		\item 对每个$g\in K(S_{1},\ldots,S_{n})$,序列$\left(e_{i}\right)_{i=1}^{n}$在$g$中可代入.
		\item $K(S_{1},\ldots,S_{n})\to K(S_{1},\ldots,S_{n})^{\mathfrak{S}_{n}},f\mapsto f\left(s_{1},\ldots,s_{n}\right)$是域同构.
	\end{enumerate}
\end{corollary}
