\section{重根的微分判别准则}

记号约定:$f\in A\left[X\right],\alpha\in A$.

\begin{proposition}{单根的微分判别法}{}
	设$\alpha$是$f$的根,则$\mult\left(f,\alpha\right)=1$ $\iff$ $\alpha$不是$\mathrm{D}f$的根.
\end{proposition}

\begin{proposition}{导数的根的重数单调递减}{}
	设$k\geq 1$且$\mult\left(f,\alpha\right)=k$.
	\begin{enumerate}
		\item $\mult\left(\mathrm{D}f,\alpha\right)\geq k-1$.
		\item 若$k\cdot 1_{A}$在$A$中可消去,则$\mult\left(\mathrm{D}f,\alpha\right)=k-1$.
	\end{enumerate}
\end{proposition}

\begin{remark}{}{}
	(2)要求$k\cdot 1_{A}$可消去是必要的.若$k>0$且$k\cdot 1_{A}=0_{A}$,则$\mult\left(X^{k},0\right)=k,\mathrm{D}X^{k}=0$,因此$0$不是$\mathrm{D}X^{k}$的$\left(k-1\right)$重根.
\end{remark}

\begin{corollary}{多重根的高阶导数判别法}{}
	设$p\in\N$且$p!\cdot 1_{A}$在$A$中可消去,则$\mult\left(f,\alpha\right)=p$ $\iff$ $\alpha$是$f,\mathrm{D}f,\ldots,\mathrm{D}^{p-1}f$的根,但不是$\mathrm{D}^{p}f$的根.
\end{corollary}
