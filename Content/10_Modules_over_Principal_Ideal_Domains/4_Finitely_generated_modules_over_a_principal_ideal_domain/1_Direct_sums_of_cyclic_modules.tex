\chapter{主理想整环上的有限生成模}

\section{循环模的直和}

记号约定:$A$是交换环,$M$是$A$-模,$m,n\geq 1$,$\left(\mathfrak{a}_{k}\right)_{k=1}^{n}$和$\left(\mathfrak{b}_{\ell}\right)_{\ell=1}^{m}$是$A$的两族理想.

\begin{proposition}{循环模的直和的外幂结构}{}
	如果$M\simeq\bigboxplus\limits_{k=1}^{n}\left(A/\mathfrak{a}_{k}\right)$,则对每个$p>0$有$\bigwedge\limits^{p}E\simeq\bigboxplus\limits_{H\in\fin_{p}\left(\left[1,n\right]\right)}A/\mathfrak{a}_{H}$.
\end{proposition}

\begin{proposition}{}{}
	设$M\simeq\bigboxplus\limits_{k=1}^{n}\left(A/\mathfrak{a}_{k}\right)$,序列$\left(\mathfrak{a}_{k}\right)_{k=1}^{n}$递增.
	\begin{enumerate}
		\item 对每个$1\leq p\leq n$有$\Ann_{A}\left(\bigwedge\limits^{p}M\right)=\mathfrak{a}_{p}$.
		\item 如果$\mathfrak{a}_{n}\neq A$,则$\bigwedge\limits^{p}\left(E\right)
		\begin{cases}
			\neq 0,&1\leq p\leq n,\\
			=0,&p>n.
		\end{cases}$
		\item 若$\left(\mathfrak{b}_{\ell}\right)_{\ell=1}^{m}$递增且$\mathfrak{b}_{m}\neq A$,$A\simeq\bigboxplus\limits_{\ell=1}^{m}\left(A/\mathfrak{b}_{k}\right)$,则序列$\left(\mathfrak{a}_{k}\right)_{k=1}^{n}$和$\left(\mathfrak{b}_{\ell}\right)_{\ell=1}^{m}$相等.
	\end{enumerate}
\end{proposition}
