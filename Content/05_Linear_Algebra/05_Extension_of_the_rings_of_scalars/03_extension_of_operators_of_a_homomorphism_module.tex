\section{同态模的算子环的延拓}

记号约定:$A$是交换环,$B$是幺环,$E,F$是$A$-模,$\rho\in\Hom_{\mathsf{Ring}}\left(A,B\right)$且$\im\left(\rho\right)\subseteq Z\left(B\right)$.

\begin{proposition}{张量积上的$A$-模结构}{}
	定义$A$在$\rho^{\ast}\left(F\right)$上的左作用为$\left(a,b\otimes y\right)\mapsto b\otimes\left(ay\right)$,则得到的$A$-模就是$\rho_{\ast}\left(\rho^{\ast}\left(F\right)\right)$.
\end{proposition}

\begin{proposition}{}{}
	存在典范的$\left(B,B\right)$-双模同态${}_{s}B_{d}\otimes_{A}\Hom_{A}\left(E,F\right)\to\Hom_{B}\left(\rho^{\ast}\left(E\right),\rho^{\ast}\left(F\right)\right)$.
\end{proposition}

\begin{proposition}{}{}
	设$\omega:{}_{s}B_{d}\otimes_{A}\Hom_{A}\left(E,F\right)\to\Hom_{B}\left(\rho^{\ast}\left(E\right),\rho^{\ast}\left(F\right)\right)$是典范的$\left(B,B\right)$-双模同态.
	\begin{enumerate}
		\item 若$_{d}B_{s}$是投射模(相对的,有限生成投射模),则$\omega$是单射(相对的,双射).
		\item 若$E$是有限生成投射模,则$\omega$是双射.
	\end{enumerate}
\end{proposition}
