\section{射影空间的定义}

记号约定:$K$是除环,$V$是左(相对的,右)$K$-向量空间,$n\geq 0$.

\begin{definition}{左射影空间,右射影空间}{}
	定义$V\setminus\left\{0\right\}$上的等价关系$\sim$如下:
	\begin{itemize}
		\item 当$V$是左$K$-向量空间时,$\forall x,y\in V\setminus\left\{0\right\}\left(x\sim y\iff\exists \lambda\in K^{\times}\left(y=\lambda x\right)\right)$.
		\item 当$V$是左$K$-向量空间时,$\forall x,y\in V\setminus\left\{0\right\}\left(x\sim y\iff\exists \lambda\in K^{\times}\left(y=x\lambda \right)\right)$.
	\end{itemize}
	定义$\mathbb{P}\left(V\right)\coloneq\left(V\setminus\left\{0\right\}\right)/\sim$.当$V$是左(相对的,右)$K$-向量空间时,称$\mathbb{P}\left(V\right)$为$V$导出的左(相对的,右)射影空间,其元素称为点.特别地,记$\mathbb{P}_{n}\left(K\right)\coloneq\mathbb{P}\left(K_{s}^{n+1}\right)$.
\end{definition}

\begin{remark}{}{}
	记$\mathcal{S}=\left\{L\subseteq V\middle|L\text{是子空间},\dim_{K}L=1\right\}$,则$\mathbb{P}\left(V\right)$典范同构于$\mathcal{S}$.
\end{remark}

\begin{definition}{射影空间的维数,射影直线,射影平面}{}
	定义$\dim_{K}\mathbb{P}\left(V\right)=
	\begin{cases}
		\dim_{K}V-1,&\dim_{K}V\text{有限},\\
		\dim_{K}V,&\dim_{K}V\text{无限}.
	\end{cases}$
	\begin{enumerate}
		\item 我们称$\dim_{K}\mathbb{P}\left(V\right)$是$V$的维数.
		\item 若$\dim_{K}\mathbb{P}\left(V\right)=1$,则称$V$是射影直线.
		\item 若$\dim_{K}\mathbb{P}\left(V\right)=2$,则称$V$是射影平面.
	\end{enumerate}
\end{definition}

\begin{remark}{记号说明}{}
	在没有特别声明的情况下,我们默认讨论左射影空间.
\end{remark}
