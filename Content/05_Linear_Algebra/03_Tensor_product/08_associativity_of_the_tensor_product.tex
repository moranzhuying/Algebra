\section{张量积的结合性}

记号约定:$A,B$是环.

\begin{proposition}{}{}
	设$E$是右$A$-模,$F$是$\left(A,B\right)$-双模,$G$是左$B$-模,则$E\otimes_{A}F$是典范的右$B$-模,$F\otimes_{B}G$是典范的左$A$-模,且有典范的$\Z$-模同构
	\begin{align*}
		\left(E\otimes_{A}F\right)\otimes_{B}G\to E\otimes_{A}\left(F\otimes_{B}G\right).
	\end{align*}
\end{proposition}

\begin{corollary}{}{}
	设$A$是交换环,$E,F,G$是$A$-模,则有典范的$A$-模同构$\left(E\otimes_{A}F\right)\otimes_{A}G\simeq E\otimes_{A}\left(F\otimes_{A}G\right)$.
\end{corollary}
