\section{有理性在线性方程组中的应用}

记号约定:

\begin{itemize}
	\item $K$是除环,$K$是除环,$K^{\prime}$是其子除环,$I,J$是集合,$\rho:K^{\prime}\hookrightarrow K$是典范嵌入.
	\item $\left(a_{i,j}\right)_{i\in I,j\in J}$是$K$的元素族,对任一$i\in I$,元素族$\left(a_{i,j}\right)_{j\in J}$有限支撑.
	\item $f:K^{I}\to K^{J},\left(x_{i}\right)_{i\in I}\mapsto\left(\sum\limits_{i\in I}a_{i,j}x_{j}\right)_{j\in J}$.
	\item $f^{\prime}:\left(K^{\prime}\right)^{I}\to \left(K^{\prime}\right)^{J},\left(x_{i}\right)_{i\in I}\mapsto\left(\sum\limits_{i\in I}a_{i,j}x_{j}\right)_{j\in J}$.
	\item $\tau_{I}:\left(K^{\prime}\right)^{I}\to\rho_{\ast}\left(K^{I}\right)$和$\tau_{J}:\left(K^{\prime}\right)^{J}\to\rho_{\ast}\left(K^{J}\right)$是典范映射.
\end{itemize}

\begin{proposition}{}{}
	$\iota_{I}\left(\ker\left(f^{\prime}\right)\right)=\rho_{\ast}\left(\ker\left(f\right)\right)\cap\im\left(\iota_{I}\right)$.进一步,$\ker\left(f\right)$由$\iota_{I}\left(\ker\left(f^{\prime}\right)\right)$生成.
\end{proposition}

\begin{proposition}{}{}
	设$\left(b_{j}\right)_{j\in J}$是$\left(K^{\prime}\right)^{I}$的元素族,则$\rho_{\ast}\left(\im\left(f\right)\right)\cap\im\left(\iota_{J}\right)=\iota_{J}\left(\im\left(f^{\prime}\right)\right)$.
\end{proposition}
