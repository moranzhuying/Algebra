\section{最小有理性除环}

记号约定:

\begin{itemize}
	\item $K$是除环,$K^{\prime},L$是其子除环.
	\item $V$是右$K$-向量空间,其$K^{\prime}$-结构是$V^{\prime}$.
\end{itemize}

\begin{proposition}{}{}
	若$K^{\prime}\subseteq L\subseteq K$,则$V^{\prime}$生成的$L$-子空间是$V$的$L$-结构.
\end{proposition}

\begin{lemma}{}{}
	设$W$是$V$的子空间,则$V$有两个$K^{\prime}$-有理子空间$W_{1},W_{2}$满足如下条件:
	\begin{itemize}
		\item $V=V_{1}\oplus V_{2}$.
		\item 在典范同构$V\simeq W_{1}\times W_{2}$的意义下,存在$g\in\Hom_{K}\left(W_{1},W_{2}\right)$使得$W=\Gr\left(g\right)$.
	\end{itemize}
\end{lemma}

\begin{proposition}{}{}
	设$x\in V$,则$K$有一个包含$K^{\prime}$且使$x$在$L$上有理的最小子除环.记该除环为$K^{\prime}\left(x\right)$.
\end{proposition}

\begin{proposition}{}{}
	设$W$是$V$的子空间,则$K$有一个包含$K^{\prime}$且使$W$是$L$-有理子空间的最小子除环.记该除环为$K^{\prime}\left(W\right)$.
\end{proposition}

\begin{proposition}{}{}
	设$V_{1},V_{2}$是$V$的子空间,各自的$K^{\prime}$-结构是$V_{1}^{\prime},V_{2}^{\prime}$,$f\in\Hom_{K}\left(V_{1},V_{2}\right)$,则存在包含$K^{\prime}$且使$f$在$L$上有理的最小子除环.记该除环为$K^{\prime}\left(f\right)$.
\end{proposition}
