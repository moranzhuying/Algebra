\chapter*{附录:伪模}

\section{伪环的幺化}

记号约定:$A$是伪环,$A^{\prime}\coloneq\Z\times A$.

\begin{definition}{伪环的幺化}{}
	在加法群$A^{\prime}$上定义乘法为$\left(\left(m,a\right),\left(n,b\right)\right)\mapsto\left(mn,mb+na+nb\right)$,则$A^{\prime}$是环,幺元是$\left(1,0\right)$.称环$A^{\prime}$是$A$的幺化.
\end{definition}

\begin{proposition}{}{}
	$\left\{0\right\}\times A$是$A^{\prime}$的双边理想,$\iota:A\to\left\{0\right\}\times A,x\mapsto\left(0,x\right)$是同构.
\end{proposition}

\begin{proposition}{}{}
	设$\varepsilon$是$A$的幺元,$e=\left(0,\varepsilon\right)$.
	\begin{enumerate}
		\item $e$是$A^{\prime}$的中心幂等元.
		\item $A=eA^{\prime}=A^{\prime}e$.
		\item $A^{\prime}=eA^{\prime}\oplus\left(1-e\right)A^{\prime},\left(1-e\right)A^{\prime}\simeq\Z$.
	\end{enumerate}
\end{proposition}
