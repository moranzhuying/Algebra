\section{饱和子集}

\begin{proposition}{饱和子集的等价条件}{equivalent-condition-of-saturated-subset}
	设$R\left\{x,y\right\}$是集合$E$上的等价关系,$A\subseteq E$,$p:E\to E/R$是典范投影,则下列陈述等价:
	\begin{enumerate}
		\item 公式$x\in A$和$R$相容.
		\item 对任意$x\in A$有$\left[x\right]_{R}\subseteq A$.
		\item $A=p^{-1}\left(p\left(A\right)\right)$.
		\item 存在$B\subseteq E/R$使得$A=p^{-1}\left(B\right)$.
	\end{enumerate}
\end{proposition}

\begin{definition}{饱和子集}{}
	设$R\left\{x,y\right\}$是集合$E$上的等价关系,$A\subseteq E$.命题(\ref{prop:equivalent-condition-of-saturated-subset})的陈述之一为真时,称$A$是相对$R$的饱和子集.
\end{definition}

\begin{proposition}{饱和子集的运算封闭性}{}
	设$R\left\{x,y\right\}$是集合$E$上的等价关系.
	\begin{enumerate}
		\item 若$\left(A_{i}\right)_{i\in I}$是$E$的一族相对$R$的饱和子集,则$\bigcup\limits_{i\in I}A_{i}$和$\bigcap\limits_{i\in I}A_{i}$都是相对$R$的饱和子集.
		\item 若$A$是$E$的相对$R$的饱和子集,则$E\setminus A$是相对$R$的饱和子集.
		\item 若$A\subseteq E$,$p:E\to E/R$是典范投影,则对$E$的任何相对$R$的饱和子集$B$都有$B\subseteq p^{-1}\left(p\left(A\right)\right)$.
	\end{enumerate}
\end{proposition}

\begin{definition}{子集的饱和化}{}
	设$R\left\{x,y\right\}$是集合$E$上的等价关系,$p:E\to E/R$是典范投影,$A\subseteq E$.我们称$p^{-1}\left(p\left(A\right)\right)$是$A$关于$R$的饱和化.
\end{definition}

\begin{proposition}{子集的饱和化的结构}{}
	设$R\left\{x,y\right\}$是集合$E$上的等价关系,$p:E\to E/R$是典范投影,$A\subseteq E$,则$p^{-1}\left(p\left(A\right)\right)=\bigcup\limits_{x\in A}\left[x\right]_{R}$.
\end{proposition}

\begin{proposition}{}{}
	设$R\left\{x,y\right\}$是集合$E$上的等价关系,$\left(X_{i}\right)_{i\in I}$是$E$的子集族,对任意$i\in I$,$A_{i}$是$X_{i}$的饱和化,则$\bigcup\limits_{i\in I}X_{i}$的饱和化是$\bigcup\limits_{i\in I}A_{i}$.
\end{proposition}
