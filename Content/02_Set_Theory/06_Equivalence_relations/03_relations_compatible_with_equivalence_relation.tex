\section{和等价关系相容的公式}

\begin{definition}{和等价关系相容的公式}{}
	设$R\left\{x,x^{\prime}\right\}$是等价关系,$P\left\{x\right\}$是公式.若对未在$P$或$R$中出现的字母$y$,有
	\begin{align*}
		P\left\{x\right\}\wedge R\left\{x,y\right\}\implies P\left\{y\right\},
	\end{align*}
	则称$P$相对于$x$和$R$相容.
\end{definition}

\begin{proposition}{每个公式都和相等关系相容}{}
	设$R\left\{x\right\}$是公式,则它和等价关系$x=y$相容.
\end{proposition}

\begin{metatheorem}{}{}
	设$R\left\{x,x^{\prime}\right\}$是集合$E$上相对$x$的等价关系,$P\left\{x\right\}$是相对$x$与$R$相容的关系,其中$x^{\prime}$不在$P$中出现.对任意$T\in E/R$,下列陈述等价:
	\begin{enumerate}
		\item $\exists x\in T\left(P\left\{x\right\}\right)$.
		\item $\forall x\in T\left(P\left\{x\right\}\right)$.
	\end{enumerate}
\end{metatheorem}

\begin{definition}{等价关系在商集上诱导的等价关系}{}
	设$R\left\{x,x^{\prime}\right\}$是集合$E$上相对$x$的等价关系,$P\left\{x\right\}$是相对$x$与$R$相容的关系.把公式
	\begin{align*}
		\exists T\in E/R\exists x\in T\left(P\left\{x\right\}\right)
	\end{align*}
	记作$P^{\prime}\left\{t\right\}$,称之为$P\left\{x\right\}$相对$R$(和$x$)在$E/R$上诱导的等价关系.
\end{definition}

\begin{proposition}{}{}
	设$R$是集合$E$上的等价关系,$P^{\prime}$是$P$在$E/R$上诱导的等价关系,则对任意$y\in E$,公式$P^{\prime}\left\{\left[y\right]_{R}\right\}$和$P\left\{y\right\}$等价.
\end{proposition}
