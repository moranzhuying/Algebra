\section{交与并的像}

\begin{proposition}{子集族的交和并在对应下的像}{}
	设$\Gamma$是集合$A$到集合$B$的对应,$\left(X_{i}\right)_{i\in I}$是$A$的子集族,则$\Gamma\left(\bigcup\limits_{i\in I}X_{i}\right)=\bigcup\limits_{i\in I}\Gamma\left(X_{i}\right),\Gamma\left(\bigcap\limits_{i\in I}X_{i}\right)\subseteq\bigcap\limits_{i\in I}\Gamma\left(X_{i}\right)$.
\end{proposition}

\begin{proposition}{交集的原像}{}
	设$f:A\to B$是映射,$\left(Y_{i}\right)_{i\in I}$是$B$的子集族,则$f^{-1}\left(\bigcap\limits_{i\in I}Y_{i}\right)=\bigcap\limits_{i\in I}f^{-1}\left(X_{i}\right)$.
\end{proposition}

\begin{corollary}{}{}
	设$f:A\to B$是单射,$\left(X_{i}\right)_{i\in I}$是$A$的子集族,$I\neq\emptyset$,则$f\left(\bigcap\limits_{i\in I}X_{i}\right)=\bigcap\limits_{i\in I}f\left(X_{i}\right)$.
\end{corollary}
