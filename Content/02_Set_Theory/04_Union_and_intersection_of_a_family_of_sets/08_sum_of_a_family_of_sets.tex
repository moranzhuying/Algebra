\section{一族集合的和}

\begin{proposition}{无交并的存在性}{}
	对任意集合族$\left(X_{i}\right)_{i\in I}$,存在集合$X$满足如下条件:
	\begin{itemize}
		\item 存在两两无交的集合族$\left(Y_{i}\right)_{i\in I}$,使得$X=\bigcup\limits_{i\in I}Y_{i}$.
		\item 对任一$i\in I$,存在从$X_{i}$到$Y_{i}$的双射.
	\end{itemize}
\end{proposition}

\begin{definition}{集合族的无交并(和)}{}
	设$\left(X_{i}\right)_{i\in I}$是集合族.把$\bigcup\limits_{i\in I}\left(X_{i}\times\left\{i\right\}\right)$记作$\coprod\limits_{i\in I}X_{i}$(或$\sum\limits_{i\in I}X_{i}$),称之为$\left(X_{i}\right)_{i\in I}$的无交并(或和).
\end{definition}

\begin{proposition}{}{}
	设$\left(X_{i}\right)_{i\in I}$是两两无交的集合族,则存在从$\bigcup\limits_{i\in I}X_{i}$到$\coprod\limits_{i\in I}X_{i}$的双射.
\end{proposition}

\begin{definition}{}{}
	设$E$是集合,$\left(X_{i}\right)_{i\in I}$是集合族.若存在双射$E\to\coprod\limits_{i\in I}X_{i}$,则称$E$是$\left(X_{i}\right)_{i\in I}$的和.
\end{definition}
