\section{Sylow子群}

记号约定:$G,G^{\prime}$是有限群,$p\in\mathbb{P}$.

\begin{definition}{Sylow $p$-群}{}
	设$H$是$G$的子群.若$H$是$p$-群,$\left(G:H\right)\nmid p$,则称$H$是$G$的Sylow $p$-群.
\end{definition}

\begin{example}{}{}
	\begin{enumerate}
		\item $p$-轮换生成的子群是$\mathfrak{S}_{p}$的Sylow $p$-子群.
		\item 设$K$是特征为$p$的有限除环,则$U_{n}\left(K\right)$是$\GL_{n}\left(K\right)$的Sylow $p$-子群.
	\end{enumerate}
\end{example}

\begin{theorem}{Sylow第一定理}{}
	$G$有Sylow $p$-子群.
\end{theorem}

\begin{corollary}{Cauchy定理}{}
	若$\left|G\right|\mid p$,则$G$有一个$p$阶的元素.
\end{corollary}

\begin{theorem}{Sylow第二定理,Sylow第三定理}{}
	\begin{enumerate}
		\item $G$的所有Sylow $p$-子群两两共轭.
		\item $G$的每个$p$-子群都包含于一个Sylow $p$-子群.
	\end{enumerate}
\end{theorem}

\begin{corollary}{}{}
	设$G$有$n_{p}$个Sylow $p$-子群,则$n_{p}\equiv 1\pmod{p}$.
\end{corollary}

\begin{corollary}{Sylow $p$-子群的正规化子的性质}{}
	设$H$是$G$的Sylow $p$-子群,$N=N_{G}\left(H\right)$.若$M$是$G$的包含$N$的子群,则$N_{G}\left(M\right)=N$.
\end{corollary}

\begin{corollary}{}{}
	设$f\in\Hom_{\mathsf{Grp}}\left(G,G^{\prime}\right)$.对$G$的每个Sylow $p$-子群$H$,$G^{\prime}$一定有一个包含$f\left(H\right)$的Sylow $p$-子群$H^{\prime}$.
\end{corollary}

\begin{corollary}{}{}
	设$H$是$G$的子群.
	\begin{enumerate}
		\item 对$H$的每个Sylow $p$-子群$P$,$G$都有一个Sylow $p$-子群$Q$使得$P=Q\cap H$.
		\item 若$H$是$G$的正规子群,$Q$是$G$的Sylow $p$-子群,则$Q\cap H$是$H$的Sylow $p$-子群.
	\end{enumerate}	
\end{corollary}

\begin{corollary}{}{}
	设$N$是$G$的正规子群,$\pi:G\twoheadrightarrow G/N$是典范满射.
	\begin{enumerate}
		\item 对$G$的每个Sylow $p$-子群$H$,$\pi\left(H\right)$是$G/N$的Sylow $p$-子群.
		\item 对$G/N$的每个Sylow $p$-子群$H^{\prime}$,存在$G$的Sylow $p$-子群$K$使得$H=\pi\left(K\right)$.
	\end{enumerate}
\end{corollary}
