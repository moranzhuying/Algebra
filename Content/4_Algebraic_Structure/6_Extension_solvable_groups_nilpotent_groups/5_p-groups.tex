\section{\texorpdfstring{$p$}{p}-群}

记号约定:$p\in\mathbb{P}$.

\begin{definition}{$p$-群}{}
	设$G$是有限群.若存在$n\in\N$使得$\left|G\right|=p^{n}$,则称$G$是$p$-群.
\end{definition}

\begin{proposition}{}{}
	\begin{enumerate}
		\item $p$-群的每个子群和商群都是$p$-群.
		\item 设$G$是$p$-群,$E$是集合,$G$左作用于$E$,则$G$-齐性空间和每个轨道的基数都是$p$的幂.
		\item 设$F,G$是$p$-群,$E$是$G$被$F$的扩张,则$E$是$p$-群.
	\end{enumerate}
\end{proposition}

\begin{example}{}{}
	\begin{enumerate}
		\item 每个交换$p$-群都同构于有限个循环群$\Z/p^{n}\Z$的乘积.
		\item 设$K$是特征为$p$的有限除环,则$SU_{n}\left(K\right)$是$p$-群.
		\item 四元数群$\left\{\pm 1,\pm i,\pm j,\pm k\right\}$是$2$-群.
	\end{enumerate}
\end{example}

\begin{proposition}{$p$-群的不动点定理}{}
	设$p$-群$G$左作用于有限集$E$.记$E^{G}=\Fix_{E}\left(G\right)$,则$\left|E^{G}\right|\equiv\left|E\right|\pmod{p}$.
\end{proposition}

\begin{corollary}{}{}
	每个非平凡的$p$-群的中心都非平凡.
\end{corollary}

\begin{theorem}{$p$-群的下中心列}{}
	设$r\geq 1$,$G$是$p^{r}$阶$p$-群,则$G$有一族递减子群$\left(G_{n}\right)_{n=1}^{r+1}$,满足下列条件:
	\begin{itemize}
		\item $G_{1}=G,G_{r+1}=\left\{e\right\}$.
		\item $\forall 1\leq k\leq r\left(\left[G,G_{k}\right]\subseteq G_{k+1}\right)$.
		\item $\left(G_{k}/G_{k+1}\right)_{k=1}^{r}$是一族$p$阶循环群.
	\end{itemize}
\end{theorem}

\begin{corollary}{$p$-群的幂零性}{}
	每个$p$-群都是幂零群.
\end{corollary}

\begin{proposition}{正规华子和极大子群}{}
	设$G$是$p$-群,$H$是$G$的真子群m则$N_{G}\left(H\right)\supset H$且$G$有一个包含$H$的正规子群$N$满足$\left(G:N\right)=p$.
\end{proposition}

\begin{corollary}{}{}
	设$G$是$p$-群,则$G$的每个指数为$p$的子群都是正规子群.
\end{corollary}
