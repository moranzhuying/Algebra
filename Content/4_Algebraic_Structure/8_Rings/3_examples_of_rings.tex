\section{环的例子}

\begin{example}{零环}{}
	设$A$是环.若$\left|A\right|=1$,则称$A$是零环.此时$\forall x\in A\left(x=0_{A}\right)$.
\end{example}

\begin{example}{整数环}{}
	$\Z$在加法和乘法下构成交换环.
\end{example}

\begin{example}{连续实值函数环}{}
	设$I$是$\R$的区间.给$\mathcal{C}\left(I,\R\right)$配备逐点的加法和乘法,则$\mathcal{C}\left(I,\R\right)$是交换环.
\end{example}

\begin{example}{卷积伪环}{}
	设$E$是$\R$上具有紧支集的连续函数组成的集合,配备逐点的加法,并且以卷积运算
	\begin{align*}
		\left(f,g\right)\mapsto\left[t\mapsto\int\limits_{-\infty}^{+\infty}f\left(s\right)g\left(t-s\right)\mathrm{d}s\right],
	\end{align*}
	作为乘法,则$E$是交换伪环.
\end{example}

\begin{example}{反环}{}
	定义环$A$的反环$A^{\op}$如下:加法群和$A$完全相同,乘法群是$A$的乘法群的反群,则$A=A^{\op}$ $\iff$ $A$是交换环.
\end{example}

\begin{example}{交换群的同态环}{}
	设$G$是加法交换群(加法记号).给$\End_{\mathsf{Grp}}\left(G\right)$配备逐点的加法,定义乘法为映射的复合,则$\End_{\mathsf{Grp}}\left(G\right)$是环.
\end{example}

\begin{example}{零平方伪环}{}
	设$A$是伪环.若$\forall x,y\in A\left(xy=0\right)$,则称$A$是零平方伪环.对诸交换群$G$(加法记号),定义乘法为
	\begin{align*}
		\left(x,y\right)\mapsto 0
	\end{align*}
	后$G$是零平方伪环,并且$G$是环$\iff$ $G=\left\{0\right\}$.
\end{example}
