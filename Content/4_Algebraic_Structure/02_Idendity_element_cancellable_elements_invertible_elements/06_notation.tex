\section{记号约定}

\begin{remark}{加法交换幺半群中的记号}{}
	设$E$是交换幺半群,采用加法记号.
	\begin{enumerate}
		\item 若$x\in E$可逆,则将$x$的逆记作$-x$.
		\item 对$x,y\in E$,定义$x-y\coloneq x+\left(-y\right)$.
		\item 对$x\in E$和$n\in\N$(进一步,当$x$可逆时$n\in\Z$),将$x$的$n$次复合记作$nx$.
	\end{enumerate}
\end{remark}

\begin{theorem}{加法交换幺半群中的广义幂运算}{}
	设$E$是交换幺半群,采用加法记号,$x,y\in E$,$m,n\in\N$(当$x,y$可逆时$m,n\in\Z$).
	\begin{enumerate}
		\item $\left(m+n\right)x=mx+nx$.
		\item $0\cdot x=0$.
		\item $1\cdot x=x$.
		\item $m\left(nx\right)=\left(mn\right)x$.
		\item 当$x$可逆时,$\left(-1\right)x=-x$.
		\item $n\left(x+y\right)=nx+ny$.
	\end{enumerate}
\end{theorem}


\begin{remark}{乘法交换幺半群中的记号}{}
	设$E$是交换幺半群,采用乘法记号.
	\begin{enumerate}
		\item 若$x\in E$可逆,则将$x$的逆记作$x^{-1}$或$\dfrac{1}{x}$.
		\item 对$x,y\in E$,定义$\dfrac{x}{y}\coloneq x/y\coloneq xy^{-1}$.
		\item 对$x\in E$和$n\in\N$(进一步,当$x$可逆时$n\in\Z$),将$x$的$n$次复合记作$x^{n}$.
	\end{enumerate}
\end{remark}

\begin{theorem}{乘法交换幺半群中的广义幂运算}{}
	设$E$是交换幺半群,采用乘法记号,$x,y\in E$,$m,n\in\Z$.
	\begin{enumerate}
		\item $x^{m+n}=x^{m}\cdot x^{n}$.
		\item $x^{0}=1$.
		\item $x^{1}=x$.
		\item $x^{mn}=\left(x^{m}\right)^{n}$.
		\item 若$x$和$y$可交换,则$\left(xy\right)^{n}=x^{n}y^{n}$.
	\end{enumerate}
\end{theorem}
