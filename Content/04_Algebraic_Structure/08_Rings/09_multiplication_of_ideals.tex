\section{理想的乘法}

记号约定:$A$是环,$\mathfrak{a},\mathfrak{b},\mathfrak{b}_{1},\ldots,\mathfrak{b}_{n}$是$A$的双边理想.

\begin{definition}{理想的乘积}{}
	定义$\mathfrak{a}\mathfrak{b}\coloneq\left\{\sum\limits_{i=1}^{m}a_{i}b_{i}\middle|m\in\N,\forall 1\leq i\leq m\left(a_{i}\in\mathfrak{a}\wedge b_{i}\in\mathfrak{b}\right)\right\}$为$\mathfrak{a}$和$\mathfrak{b}$的乘积.
\end{definition}

\begin{proposition}{}{}
	$\mathfrak{a}\mathfrak{b}$是$A$的双边理想,并且$\mathfrak{a}\mathfrak{b}\subseteq\mathfrak{a}\cap\mathfrak{b}$.
\end{proposition}

\begin{proposition}{双边理想成绩的性质}{}
	记$\Phi$是$A$的双边理想组成的集合,按照理想乘法赋予其乘法,通过理想的和赋予其加法.
	\begin{enumerate}
		\item $\Phi$对乘法构成幺半群,其加法对乘法分配.
		\item 若$A$是交换环,则$\Phi$对乘法构成交换幺半群.
	\end{enumerate}
\end{proposition}

\begin{proposition}{理想的和的性质}{}
	若$\forall 1\leq i\leq n\left(\mathfrak{a}+\mathfrak{b}_{i}=A\right)$,则$A=\mathfrak{a}+\prod\limits_{i=1}^{n}\mathfrak{b}_{i}=\mathfrak{a}+\bigcap\limits_{i=1}^{n}\mathfrak{b}_{i}$.
\end{proposition}

\begin{proposition}{两两互素理想的交}{}
	设$\forall 1\leq i\neq j\leq n\left(\mathfrak{b}_{i}+\mathfrak{b}_{j}=A\right)$,则$\bigcap\limits_{i=1}^{n}\mathfrak{b}_{i}=\sum\limits_{\sigma\in\mathfrak{S}_{n}}\prod\limits_{i=1}^{m}\mathfrak{b}_{\sigma\left(i\right)}$.特别地,若$A$是交换环,则$\bigcap\limits_{i=1}^{n}\mathfrak{b}_{i}=\prod\limits_{i=1}^{n}\mathfrak{b}_{i}$.
\end{proposition}
