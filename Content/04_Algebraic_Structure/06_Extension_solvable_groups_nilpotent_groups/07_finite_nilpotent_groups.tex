\section{有限幂零群}

记号约定:$G$是有限群,$p\in\mathbb{P}$.

\begin{theorem}{有限幂零群的等价刻画}{}
	下列陈述等价:
	\begin{enumerate}
		\item $G$是幂零群.
		\item $G$同构于其各阶Sylow $p$-子群的乘积.
		\item 若$p\mid\left|G\right|$,$G$的每个Sylow $p$-子群都是正规子群.
	\end{enumerate}
\end{theorem}

\begin{remark}{}{}
	\begin{enumerate}
		\item $G$有一个正规Sylow $p$-子群$\iff$ $G$只有一个Sylow $p$-子群$\iff$ $G$的每个Sylow $p$-子群都正规.
		\item 设$G$是幂零群,$I$是$\left|G\right|$的因子组成的集合,则$G$可以唯一的表示成$\prod\limits_{i\in I}G_{i}$,其中$G_{i}$是$G$唯一的Sylow $i$-子群.进一步,若$G$是交换群,则我们得到了$G$的准素分解.
	\end{enumerate}
\end{remark}

\begin{example}{}{}
	$\mathfrak{S}_{3}$是$6$阶群,有一个$3$阶正规Sylow $3$-子群$\mathfrak{A}_{3}$,$3$个$2$阶非正规的Sylow $2$-子群,则$\mathfrak{S}_{3}$是幂零群.
\end{example}
