\section{自由对象之间的关系}

记号约定:设$X$是集合.考虑如下的典范映射:
\begin{align*}
	&\lambda:\Mag\left(X\right)\to\Mo\left(X\right)\\
	&\mu:\Mo\left(X\right)\to F\left(X\right)\\
	&\nu:\Mo\left(X\right)\to\N^{\left(X\right)}\\
	&\pi:F\left(X\right)\to\Z^{\left(X\right)}\\
	&\iota:\N^{\left(X\right)}\to\Z^{\left(X\right)}
\end{align*}

\begin{proposition}{}{}
	以下图表交换.
	\begin{center}
		\begin{tikzcd}
			{\Mag\left(X\right)} & {\Mo\left(X\right)} & {\N^{\left(X\right)}} \\
			& {F\left(X\right)} & {\Z^{\left(X\right)}}
			\arrow["\lambda", from=1-1, to=1-2]
			\arrow["\nu", from=1-2, to=1-3]
			\arrow["\mu"', from=1-2, to=2-2]
			\arrow["\pi"', from=2-2, to=2-3]
			\arrow["\iota", from=2-3, to=1-3]
		\end{tikzcd}
	\end{center}
\end{proposition}

\begin{proposition}{长度与同态}{}
	\begin{enumerate}
		\item 若$\omega\in\Mag\left(X\right)$,则$\left|\nu\left(\omega\right)\right|=\ell\left(\omega\right)$.
		\item 若$u=\prod\limits_{i=1}^{n}x_{i}\in\Mo\left(X\right)$,则$\nu\left(u\right)=\sum\limits_{i=1}^{n}\delta_{x_{i}}$.
	\end{enumerate}
\end{proposition}

\begin{proposition}{阿贝尔化的核}{}
	$\mu$是满射.
\end{proposition}

\begin{proposition}{}{}
	设$\omega,\omega^{\prime}\in\Mo\left(X\right)$,其中$\omega=\prod\limits_{i=1}^{n}x_{i},\omega^{\prime}=\prod\limits_{i=1}^{m}x_{i}^{\prime}$.下列陈述等价:
	\begin{enumerate}
		\item $\ell\left(\omega\right)=\ell\left(\omega^{\prime}\right)$.
		\item $m=n$且存在$\sigma\in\mathfrak{S}_{n}$使得$\forall 1\leq i\leq n\left(x_{i}^{\prime}=x_{\sigma\left(i\right)}\right)$.
	\end{enumerate}
\end{proposition}


\begin{proposition}{子对象与原像}{}
	设$S\subseteq X$,则:
	\begin{enumerate}
		\item 在同构意义下,$\Mag\left(S\right)=\lambda^{-1}\left(\Mo\left(S\right)\right),\Mo\left(S\right)=\nu^{-1}\left(\N^{\left(S\right)}\right)$.
		\item 若$\left(S_{i}\right)_{i\in I}$是$X$的一族子集,$\bigcap\limits_{i\in I}S_{i}=S$,则$\Mag\left(S\right)=\bigcap\limits_{i\in I}\Mag\left(S_{i}\right),\Mo\left(S\right)=\bigcap\limits_{i\in I}\Mo\left(S_{i}\right)$.
	\end{enumerate}	
\end{proposition}
