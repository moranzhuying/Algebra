\section{有序群之间的保序同态同态}

记号约定:$G,H$是有序群.

\begin{proposition}{有序群同态的等价条件}{equivalent-condition-of-ordered-group-homomorphism}
	设$f\in\Hom_{\mathsf{Grp}}\left(G,H\right)$,则$f$是保序映射$\iff$ $f\left(G_{\geqslant 0}\right)\subseteq H_{\geqslant 0}$.
\end{proposition}

\begin{definition}{有序群同态}{}
	设$f\in\Hom_{\mathsf{Grp}}\left(G,H\right)$.若$f$满足命题(\ref{prop:equivalent-condition-of-ordered-group-homomorphism})的等价条件之一,则称$f$是有序群同态.
\end{definition}

\begin{remark}{}{}
	\begin{enumerate}
		\item 若$H$是$G$的有序子群,则典范嵌入$H\hookrightarrow G$是有序群同态.
		\item 若$\left(G_{i}\right)_{i\in I}$是一族有序群,$\prod\limits_{i\in I}G_{i}$配备逐点偏序,则$\left(\pr_{i}\right)_{i\in I}$是一族有序群同态.
	\end{enumerate}
\end{remark}

\begin{proposition}{有序群同构的等价条件}{equivalent-condition-of-ordered-group-isomorphism}
	设$f\in\Isom_{\mathsf{Grp}}\left(G,H\right)$,则$f$是保序群同构$\iff$ $f\left(G_{\geq 0}\right)=H_{\geq 0}$ $\iff$ $f$和$f^{-1}$都是保序同构.
\end{proposition}

\begin{definition}{有序群同构}{}
	设$f\in\Isom_{\mathsf{Grp}}\left(G,H\right)$是群同构.若$f$满足命题(\ref{prop:equivalent-condition-of-ordered-group-isomorphism})的条件之一,则称$f$是有序群同构.
\end{definition}

\begin{remark}{}{}
	如果$f$是保序同构但$f^{-1}$不是保序同构,那么$f$不一定是有序群同构.
	\begin{itemize}
		\item 设$G$和$H$作为群都是$\Z$的加法群,$f=\id_{G}$,但是$G_{geq 0}=\Z_{\geq 0},H_{\geq 0}=2\Z$,那么$f$不是有序群同构.
		\item 设$K=\F_{2}\left(X\right),A_{1}=\F_{2}[X],A_{2}=\F_{2}[X^{2},X^{3}]$.令$G$和$H$分别是$A_{1}$和$A_{2}$在$K^{\times}$上定义的有序群,那么$f$不是有序群同构.
	\end{itemize}
\end{remark}
