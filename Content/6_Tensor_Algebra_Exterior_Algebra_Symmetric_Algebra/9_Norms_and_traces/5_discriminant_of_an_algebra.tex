\section{代数的判别式}

记号约定:$A$作为$K$-模具备一个基数为$n$的基集.

\begin{definition}{序列的判别式}{}
	设$\left(x_{i}\right)_{i=1}^{n}$是$A$中的序列.我们称$\left(\Tr_{A/K}\left(x_{i}x_{j}\right)\right)_{\substack{1\leq i\leq n\\1\leq j\leq n}}$的行列式为$\left(x_{i}\right)_{i=1}^{n}$关于$K$的判别式,将其记作$D_{A/K}\left(x_{1},\ldots,x_{n}\right)$.
\end{definition}

\begin{proposition}{结构常数和判别式}{}
	设$\left(e_{i}\right)_{i=1}^{n}$是$A$的$K$-基,结构常数为$\left(c_{i,j,k}\right)_{1\leq i,j,k\leq n}$,则$D_{A/K}\left(e_{1},\ldots,e_{n}\right)=\det\left(\sum\limits_{k=1}^{n}\sum\limits_{s=1}^{n}c_{i,j,k}c_{k,s,s}\right)_{\substack{1\leq i\leq n\\1\leq j\leq n}}$.
\end{proposition}

\begin{proposition}{判别式的基变换公式}{}
	设$\left(x_{i}\right)_{i=1}^{n},\left(y_{i}\right)_{i=1}^{n}$是$A$中的序列,$\boldsymbol{M}=\left(m_{i,j}\right)_{\substack{1\leq i\leq n\\1\leq j\leq n}}\in\Mat_{n}\left(K\right)$满足$\forall 1\leq i\leq n\left(x_{i}=\sum\limits_{j=1}^{n}m_{i,j}y_{j}\right)$,则
	\begin{align*}
		D_{A/K}\left(x_{1},\ldots,x_{n}\right)=\left(\det\left(\boldsymbol{M}\right)\right)^{2}D_{A/K}\left(y_{1},\ldots,y_{n}\right).
	\end{align*}
\end{proposition}

\begin{definition}{判别式理想}{}
	设设$\left(x_{i}\right)_{i=1}^{n}$是$A$中的序列.我们称$D_{A/K}\left(x_{1},\ldots,x_{n}\right)$在$K$上生成的主理想为$A$在$K$上的判别式理想,将其记作$\Delta_{A/K}$.
\end{definition}

\begin{example}{}{}
	\begin{enumerate}
		\item 设$A$是$K$上的$\left(\alpha,\beta\right)$型二次代数,典范基为$\left(e_{i}\right)_{i=1}^{2}$,则$D_{A/K}\left(e_{1},e_{2}\right)=\beta^{2}+4\alpha$.
		\item 设$A=K\left[X\right]/\langle X^{3}+pX+q\rangle,x=X+\langle X^{3}+pX+q\rangle$,则$\left(x^{k-1}\right)_{k=1}^{3}$是$K$的基,并且
		\begin{align*}
			D_{A/K}\left(1,x,x^{2}\right)=-4p^{3}-27q^{2}.
		\end{align*}
		\item 设$A$是$K$上的$\left(\alpha,\beta,\gamma\right)$型四元数代数,典范基为$\left(1,i,j,k\right)$,则$D_{A/K}\left(1,i,j,k\right)=-16\gamma^{2}\left(\beta^{2}+4\alpha\right)$.
		\item 设$A=\Mat_{n}\left(K\right)$,其典范基为$\left(\boldsymbol{E}_{i,j}\right)_{\substack{1\leq i\leq n\\1\leq j\leq n}}$,则$D_{A/K}\left(\boldsymbol{E}_{1,1},\ldots,\boldsymbol{E}_{n,n}\right)=\pm n^{n^{2}}$.
	\end{enumerate}
\end{example}
