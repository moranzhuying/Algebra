\section{幺模群\texorpdfstring{$\SL_{n}\left(A\right)$}{SL(n,A)}}

记号约定:$A$是交换环,$K$是域,$n>0$.

\begin{proposition}{行列式给出群满同态}{}
	$\det:\GL_{n}\left(A\right)\to A^{\times},\boldsymbol{A}\mapsto\det\left(\boldsymbol{A}\right)$既是群同态又是满射.
\end{proposition}

\begin{definition}{幺模群(特殊线性群)}{}
	记$\det:\GL_{n}\left(A^{}\right)\to A^{\times},\boldsymbol{A}\mapsto\det\left(\boldsymbol{A}\right)$.
	\begin{enumerate}
		\item 我们称$\SL_{n}\left(A\right)\coloneq\ker\left(\det\right)$是一般线性群(或特殊线性群).
		\item 我们称$\SL_{n}\left(A\right)$的元素为$A$上的$n$阶幺模矩阵.
	\end{enumerate}
\end{definition}

\begin{proposition}{$\SL_{n}\left(A\right)$的生成集}{}
	$\SL_{n}\left(A\right)$由$\left\{\boldsymbol{B}_{i,j}\left(\lambda\right)\in\Mat_{n}\left(A\right)\middle|1\leq i\neq j\leq n,\lambda\in K\right\}$生成.
\end{proposition}

\begin{corollary}{}{}
	当$n\neq 2$或$K\neq\mathbb{F}_{2}$时,$\SL_{n}\left(K\right)$是$\GL_{n}\left(K\right)$的导群.
\end{corollary}

\begin{remark}{}{}
	\begin{enumerate}
		\item $\GL_{2}\left(\mathbb{F}_{2}\right)=\SL_{2}\left(\mathbb{F}_{2}\right)$.这是一个$6$阶可解群,它的交换子群的指数为$2$.
		\item 进一步,$\SL_{n}\left(K\right)$可以由$\left\{\boldsymbol{B}_{1,2}\left(\lambda\right)\in\Mat_{n}\left(K\right)\middle|\lambda\in K\right\}\cup\left\{\boldsymbol{P}_{\tau_{i,i+1}}\in\Mat_{n}\left(K\right)\middle|1\leq i\leq i-1\right\}$生成.
	\end{enumerate}
\end{remark}
