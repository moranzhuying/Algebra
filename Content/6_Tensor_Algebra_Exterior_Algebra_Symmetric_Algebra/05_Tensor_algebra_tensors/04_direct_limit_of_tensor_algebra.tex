\section{张量代数的正向极限}

记号约定:
\begin{itemize}
	\item $\left(\left(A_{\alpha}\right)_{\alpha\in I},\left(\phi_{\beta\alpha}\right)_{\alpha,\beta\in I}\right)$是交换环的正系统,$A=\varinjlim A_{\alpha}$.
	\item $\left(\left(M_{\alpha}\right)_{\alpha\in I},\left(f_{\beta\alpha}\right)_{\alpha,\beta\in I}\right)$是$A_{\alpha}$-模的正系统,$M=\varinjlim M_{\alpha}$.
	\item 对每个$\alpha,\beta\in I$且$\alpha\leq\beta$,记$f_{\beta\alpha}^{\prime}=\mathsf{T}\left(f_{\beta\alpha}\right)$.
	\item 对每个$\alpha\in I$,记$f_{\alpha}^{\prime}=\mathsf{T}\left(f_{\alpha}\right)$.
\end{itemize}

\begin{proposition}{}{}
	$\left(\left(\mathsf{T}\left(M_{\alpha}\right)\right)_{\alpha\in I},\left(f_{\beta\alpha}^{\prime}\right)_{\alpha,\beta\in I}\right)$是$A_{\alpha}$-代数的正系统.
\end{proposition}

\begin{proposition}{张量代数和正向极限的交换性}{}
	$\varinjlim f_{\alpha}^{\prime}$是$\N$型分次$A$-代数同构.
\end{proposition}
