\section{有限秩自由模上的自同构的刻画}

记号约定:$A$是交换环,$M$是有限维自由$A$-模.

\begin{theorem}{}{}
	设$u\in\End_{A}\left(M\right)$,则$u$是双射$\iff$ $u$是右可逆映射$\iff$ $u$是左可逆映射$\iff$ $u$是满射$\iff$ $\det\left(u\right)\in A^{\times}$.
\end{theorem}

\begin{proposition}{}{}
	设$u\in\End_{A}\left(M\right)$,则$u$是单射$\iff$ $\det\left(u\right)$不是$A$的零因子.
\end{proposition}

\begin{remark}{}{}
	当$A$是域时,$u$是双射$\iff$ $u$是右可逆映射$\iff$ $u$是左可逆映射$\iff$ $u$是满射$\iff$ $u$是单射$\iff$ $\det\left(u\right)\neq 0_{A}$.
\end{remark}
