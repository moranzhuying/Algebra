\section{矩阵代数}

记号约定:$A$是交换环,$B$是$A$-含幺结合代数,$n\geq 1$.

\begin{proposition}{含幺结合代数上的矩阵代数}{}
	定义$A$在$\mathbf{M}_{n}\left(B\right)$上的左作用为$\left(\alpha,\left(b_{i,j}\right)_{\substack{1\leq i\leq n\\ 1\leq j\leq n}}\right)\mapsto\left(\alpha b_{i,j}\right)_{\substack{1\leq i\leq n\\ 1\leq j\leq n}}$,则$\mathbf{M}_{n}\left(B\right)$是$A$-含幺结合代数,其乘法为矩阵乘法.
\end{proposition}

\begin{corollary}{交换环上的矩阵代数}{}
	$\mathbf{M}_{n}\left(A\right)$是$A$-含幺结合代数,它有典范基$\left(\boldsymbol{E}_{i,j}\right)_{\substack{1\leq i\leq n\\ 1\leq j\leq n}}$,它满足
	\begin{gather*}
		\forall 1\leq i,j,k,\ell\leq n\left(\boldsymbol{E}_{i,j}\boldsymbol{E}_{k,\ell}=\delta_{j,k}\boldsymbol{E}_{i,\ell}\right),\\
		\boldsymbol{I}_{n\times n}=\sum\limits_{i=1}^{n}\boldsymbol{E}_{i,i}.
	\end{gather*}
\end{corollary}
