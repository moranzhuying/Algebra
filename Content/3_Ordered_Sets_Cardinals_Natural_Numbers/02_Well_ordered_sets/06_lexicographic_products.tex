\section{字典序乘积}

\begin{definition}{字典序}{}
	设$\left(E_{i}\right)_{i\in I}$是一族偏序集,$I$是良序集,$E=\prod\limits_{i\in I}E_{i}$.对任意$x,y\in E$,记$J_{x,y}=\left\{i\in I\middle|\pr_{i}\left(x\right)\neq\pr_{i}\left(y\right)\right\}$.把公式
	\begin{align*}
		x\in E\wedge y\in E\wedge\forall i\in I\left(i=\min J_{x,y}\implies\pr_{i}\left(x\right)<\pr_{i}\left(y\right)\right)
	\end{align*}
	称为$E$上的字典序关系,它定义的$E$上的偏序称为$E$上的字典序,配备字典序的$E$称为$\left(E_{i}\right)_{i\in I}$的字典序乘积.
\end{definition}

\begin{remark}{}{}
	当$\left(E_{i}\right)_{i\in I}$是一族全序集时,$E$上的字典序是全序.
\end{remark}
