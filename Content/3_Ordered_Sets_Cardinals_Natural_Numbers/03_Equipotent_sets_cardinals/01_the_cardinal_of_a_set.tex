\chapter{等势集,基数}

\section{集合的基数}

\begin{definition}{等势}{}
	设$X,Y$是集合.如果$\Isom_{\mathsf{Set}}\left(X,Y\right)\neq\emptyset$,则称$X$和$Y$等势.我们把公式``$X$和$Y$等势''记作$\Eq\left(X,Y\right)$.
\end{definition}

\begin{remark}{}{}
	可以验证$\Eq\left(X,Y\right)$是等价关系.
\end{remark}

\begin{definition}{集合的基数}{}
	设$X$是集合.我们把集合$\tau_{Z}\left(\Eq\left(X,Z\right)\right)$记作$\left|X\right|$(或$\# X$),称之为$X$的基数.
\end{definition}

\begin{proposition}{集合等势$\iff$集合的基数相同}{}
	设$X,Y$是集合,则$X$和$Y$等势$\iff$ $\left|X\right|=\left|Y\right|$.
\end{proposition}

\begin{example}{}{}
	\begin{enumerate}
		\item 和$\emptyset$等势的集合只有$\emptyset$.我们把$\left|\emptyset\right|$记作$0$.
		\item 任意两个由单点集组成的集合都等势.我们把$\left|\left\{\emptyset\right\}\right|$记作$1$.
		\item 我们把$\left|\left\{\emptyset,\left\{\emptyset\right\}\right\}\right|$记作$2$.每一个由两个互异的元素组成的集合的基数都是$2$.
		\item 可分的Hillbert空间与$\R$等势.
	\end{enumerate}
\end{example}
