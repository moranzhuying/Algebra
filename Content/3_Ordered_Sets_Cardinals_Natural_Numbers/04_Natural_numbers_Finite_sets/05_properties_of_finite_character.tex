\section{有限特征的性质}

\begin{definition}{有限特征集合族}{}
	设$E$是集合.
	\begin{enumerate}
		\item 设$\mathfrak{S}\subseteq\mathcal{P}\left(E\right)$.若对每个集合$X$都有$X\in\mathfrak{S}\iff\fin\left(X\right)\subseteq\mathfrak{S}$,则称$\mathfrak{S}$是有限特征的.
		\item 设$X\subseteq E$,$P\left\{X\right\}$是公式.若$E$的所有满足$P\left\{X\right\}$的子集组成的集合是有线特征的,则称$P\left\{X\right\}$是有线特征的.
	\end{enumerate}
\end{definition}

\begin{example}{}{}
	\begin{enumerate}
		\item 偏序集$E$的所有全序子集构成的集合是有限特征的.
		\item 一个环上的左模的线性无关集组成的集合是有限特征的.一个除环的扩张的所有代数无关的子集构成的集合是有限特征的.
		\item 一个环上的模的所有子模组成的集合不是有限特征的.
	\end{enumerate}
\end{example}

\begin{theorem}{Teichm\"{u}ller-Turkey引理}{}
	设$E$是集合,$\mathfrak{S}\subseteq\mathcal{P}\left(E\right)$且有限特征.为$\mathfrak{S}$配备包含偏序,则该集合有极大元.
\end{theorem}
