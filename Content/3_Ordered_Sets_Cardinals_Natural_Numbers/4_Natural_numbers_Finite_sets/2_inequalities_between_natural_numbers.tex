\section{自然数之间的不等式}

\begin{proposition}{}{}
	设$n$是自然数.
	\begin{enumerate}
		\item 所有满足$\mathfrak{a}\leq n$的基数$\mathfrak{a}$都是自然数.
		\item 若$n\neq 0$,则存在唯一的自然数$m$使得$n=m+1$,并且对每个基数$\mathfrak{a}$都有$\mathfrak{a}<n\iff\mathfrak{a}\leq m$.
	\end{enumerate}
\end{proposition}

\begin{corollary}{}{}
	一个有限集的每个子集都是有限集.
\end{corollary}

\begin{corollary}{}{}
	设$E$是集合.
	\begin{enumerate}
		\item 若$E$是有限集,$X$是$E$的真子集,则$\left|X\right|<\left|E\right|$.
		\item 若$E$的每个真子集都是有限集,则$E$是有限集.
	\end{enumerate}
\end{corollary}

\begin{corollary}{}{}
	若$E,F$是有限集,$f\in\Hom_{\mathsf{Set}}\left(E,F\right)$,则$\im\left(f\right)$是有限集.
\end{corollary}

\begin{corollary}{}{}
	设$E,F$是有限集且$\left|E\right|=\left|F\right|$,$f\in\Hom_{\mathsf{Set}}\left(E,F\right)$,则$f$是单射$\iff$ $f$是满射$\iff$ $f$是双射.
\end{corollary}
