\section{有序集的有限子集}

\begin{definition}{集合的有限子集}{}
	设$E$是集合.我们把$E$的有限子集组成的集合记作$\fin\left(E\right)$.
\end{definition}

\begin{proposition}{}{}
	设$E$是配备了预序(或偏序)的有限集,则对每个$X\in\fin\left(E\right)\setminus\left\{\emptyset\right\}$,
	\begin{enumerate}
		\item 若$E$是上定向的,则$X$在$E$中有上界.
		\item 若$E$是格,则$X$在$E$中有上确界和下确界(换言之,$X$是$E$的子格).
		\item 若$E$是全序集,则$X$有最大元和最小元.
	\end{enumerate}
\end{proposition}

\begin{corollary}{}{}
	每一个配备了全序的有限集都是具有最大元的全序集.
\end{corollary}

\begin{corollary}{}{}
	每一个配备了偏序的有限集都是具有极大元的偏序集.
\end{corollary}
