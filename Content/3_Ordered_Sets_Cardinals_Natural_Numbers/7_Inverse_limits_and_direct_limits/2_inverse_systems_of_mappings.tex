\section{映射的逆向系统}

\begin{proposition}{映射的逆向极限的泛性质}{}
	设$I$是预序集,$\left(\left(E_{\alpha}\right)_{\alpha\in I},\left(f_{\alpha,\beta}\right)_{\alpha,\beta\in I}\right)$是集合的逆向系统,$\left(f_{\alpha}\right)_{\alpha\in I}$是典范映射族,$F$是集合,对任意$\alpha\in I$有
	\begin{align*}
		u_{\alpha}\in\Hom_{\mathsf{Set}}\left(F,E_{\alpha}\right).
	\end{align*}
	如果$\forall\alpha,\beta\in I\left(\alpha\leq\beta\implies f_{\alpha,\beta}\circ u_{\beta}=u_{\alpha}\right)$,则
	\begin{enumerate}
		\item 存在唯一的$u\in\Hom_{\mathsf{Set}}\left(F,\varprojlim E_{\alpha}\right)$使得$\forall\alpha\in I\left(u_{\alpha}=f_{\alpha}\circ u\right)$.
		\item $u$是单射$\iff$ $\forall y,z\in F\left(y\neq z\implies\exists\alpha\in I\left(u_{\alpha}\left(y\right)\neq u_{\alpha}\left(z\right)\right)\right)$.
	\end{enumerate}
\end{proposition}

上述命题用交换图描述如下:
\begin{center}
	\begin{tikzcd}
		&& F && \\
		\\
		&& {\varprojlim E_{\alpha}} \\
		{E_{\beta}} &&&& {E_{\alpha}}
		\arrow["{\exists!u}", dashed, from=1-3, to=3-3]
		\arrow["{u_{\beta}}"', from=1-3, to=4-1]
		\arrow["{u_{\alpha}}", from=1-3, to=4-5]
		\arrow["{f_{\beta}}"', from=3-3, to=4-1]
		\arrow["{f_{\alpha}}", from=3-3, to=4-5]
		\arrow["{f_{\alpha,\beta}}", from=4-1, to=4-5]
	\end{tikzcd}
\end{center}

\begin{corollary}{映射的逆向极限的泛性质}{universal-property-of-a-family-of-mappings}
	设$\left(\left(E_{\alpha}\right)_{\alpha\in I},\left(f_{\alpha,\beta}\right)_{\alpha,\beta\in I}\right),\left(\left(F_{\alpha}\right)_{\alpha\in I},\left(g_{\alpha,\beta}\right)_{\alpha,\beta\in I}\right)$是集合的逆向系统,$\left(f_{\alpha}\right)_{\alpha\in I},\left(g_{\alpha}\right)_{\alpha\in I}$分别是二者的典范映射族.若对每个$\alpha\in I$,存在$u_{\alpha}\in\Hom_{\mathsf{Set}}\left(E_{\alpha},F_{\alpha}\right)$,对任意$\beta\in I$,当$\alpha\leq\beta$时下图交换:
	\begin{center}
		\begin{tikzcd}
			{E_{\beta}} & {F_{\beta}} \\
			{E_{\alpha}} & {F_{\alpha}}
			\arrow["{u_{\beta}}", from=1-1, to=1-2]
			\arrow["{f_{\alpha,\beta}}"', from=1-1, to=2-1]
			\arrow["{g_{\alpha,\beta}}", from=1-2, to=2-2]
			\arrow["{u_{\alpha}}"', from=2-1, to=2-2]
		\end{tikzcd}
	\end{center}
	则存在唯一的$u\in\Hom_{\mathsf{Set}}\left(\varprojlim E_{\alpha},\varprojlim F_{\alpha}\right)$,对任意$\alpha\in I$下图交换:
	\begin{center}
		\begin{tikzcd}
			{\varprojlim E_{\alpha}} & {\varprojlim F_{\alpha}} \\
			{E_{\alpha}} & {F_{\alpha}}
			\arrow["{\exists!u}", dashed, from=1-1, to=1-2]
			\arrow["{f_{\alpha}}"', from=1-1, to=2-1]
			\arrow["{g_{\alpha}}", from=1-2, to=2-2]
			\arrow["{u_{\alpha}}"', from=2-1, to=2-2]
		\end{tikzcd}
	\end{center}
\end{corollary}

\begin{definition}{映射的逆向系统和你极限}{}
	记号同推论(\cref{cor:universal-property-of-a-family-of-mappings}).
	\begin{enumerate}
		\item 我们称$\left(u_{\alpha}\right)_{\alpha\in I}$是$\left(\left(E_{\alpha}\right)_{\alpha\in I},\left(f_{\alpha,\beta}\right)_{\alpha,\beta\in I}\right)$到$\left(\left(F_{\alpha}\right)_{\alpha\in I},\left(g_{\alpha,\beta}\right)_{\alpha,\beta\in I}\right)$的映射逆向系统.
		\item 我们把$u$记作$\varprojlim\limits_{I}u_{\alpha}$(或$\varprojlim u_{\alpha}$),称之为$\left(u_{\alpha}\right)_{\alpha\in I}$的逆向极限.
	\end{enumerate}
\end{definition}

\begin{corollary}{映射的逆向极限的函子性}{}
	设$\left(\left(E_{\alpha}\right)_{\alpha\in I},\left(f_{\alpha,\beta}\right)_{\alpha,\beta\in I}\right),\left(\left(F_{\alpha}\right)_{\alpha\in I},\left(g_{\alpha,\beta}\right)_{\alpha,\beta\in I}\right),\left(\left(G_{\alpha}\right)_{\alpha\in I},\left(h_{\alpha,\beta}\right)_{\alpha,\beta\in I}\right)$是集合的逆向系统.若
	\begin{itemize}
		\item 三者各自的典范映射族分别是$\left(f_{\alpha}\right)_{\alpha\in I},\left(g_{\alpha}\right)_{\alpha\in I},\left(h_{\alpha}\right)_{\alpha\in I}$,
		\item $\left(u_{\alpha}\right)_{\alpha\in I}$和$\left(v_{\alpha}\right)_{\alpha\in I}$分别是$\left(E_{\alpha}\right)_{\alpha\in I}$到$\left(F_{\alpha}\right)_{\alpha\in I}$和$\left(F_{\alpha}\right)_{\alpha\in I}$到$\left(G_{\alpha}\right)_{\alpha\in I}$的映射逆向系统,
	\end{itemize}
	则$\left(v_{\alpha}\circ u_{\alpha}\right)_{\alpha\in I}$是$\left(E_{\alpha}\right)_{\alpha\in I}$到$\left(G_{\alpha}\right)_{\alpha\in I}$的逆向系统,并且下列图表交换:
	\begin{center}
		\begin{tikzcd}
			{\varprojlim E_{\alpha}} && {\varprojlim F_{\alpha}} \\
			& {\varprojlim G_{\alpha}}
			\arrow["{\varprojlim u_{\alpha}}", from=1-1, to=1-3]
			\arrow["{\varprojlim \left(v_{\alpha}\circ u_{\alpha}\right)}"', from=1-1, to=2-2]
			\arrow["{\varprojlim v_{\alpha}}", from=1-3, to=2-2]
		\end{tikzcd}
	\end{center}
\end{corollary}

\begin{proposition}{子集的逆向系统}{}
	设$\left(\left(E_{\alpha}\right)_{\alpha\in I},\left(f_{\alpha,\beta}\right)_{\alpha,\beta\in I}\right)$是集合的逆向系统,对任意$\alpha\in I$,$M_{\alpha}\subseteq E_{\alpha}$且$\forall\beta\in I\left(\alpha\leq\beta\implies f_{\alpha,\beta}\left(M_{\beta}\right)\subseteq M_{\alpha}\right)$.
	
	若对任意$\alpha,\beta\in I$且$\alpha\leq\beta$,$g_{\alpha,\beta}\in\Hom_{\mathsf{Set}}\left(M_{\beta},M_{\alpha}\right)$且$\Gr\left(g_{\alpha,\beta}\right)=\Gr\left(f_{\alpha,\beta}|_{M_{\beta}}\right)$,则$\varprojlim M_{\alpha}=\left(\varprojlim E_{\alpha}\right)\cap\prod\limits_{\alpha\in I}M_{\alpha}$.
\end{proposition}

\begin{proposition}{纤维的逆向极限}{inverse-limit-of-fibre}
	设$\left(\left(E_{\alpha}\right)_{\alpha\in I},\left(f_{\alpha,\beta}\right)_{\alpha,\beta\in I}\right),\left(\left(F_{\alpha}\right)_{\alpha\in I},\left(g_{\alpha,\beta}\right)_{\alpha,\beta\in I}\right)$是集合的逆向系统,而$\left(u_{\alpha}\right)_{\alpha\in I}$是$\left(E_{\alpha}\right)_{\alpha\in I}$到$\left(F_{\alpha}\right)_{\alpha\in I}$的映射逆向系统,并记$u=\varprojlim u_{\alpha}$,则对任意$\left(y_{\alpha}\right)_{\alpha\in I}\in\varprojlim F_{\alpha}$,$\left(u_{\alpha}^{-1}\left(\left\{y_{\alpha}\right\}\right)\right)_{\alpha\in I}$是$\left(E_{\alpha}\right)_{\alpha\in I}$的子集逆向系统,并且
	\begin{align*}
		u^{-1}\left(\left\{\left(y_{\alpha}\right)_{\alpha\in I}\right\}\right)=\varprojlim u_{\alpha}^{-1}\left(\left\{y_{\alpha}\right\}\right).
	\end{align*}
\end{proposition}

\begin{corollary}{}{}
	记号同命题(\cref{prop:inverse-limit-of-fibre}).若$\left(u_{\alpha}\right)_{\alpha\in I}$是一族单射(相应的,双射),则$u$也是单射(相应的,双射).
\end{corollary}

\begin{proposition}{共尾限制}{}
	设$I$是预序集,$\left(\left(E_{\alpha}\right)_{\alpha\in I},\left(f_{\alpha,\beta}\right)_{\alpha,\beta\in I}\right)$是集合的逆向系统,$J$是$I$的共尾子集.
	\begin{enumerate}
		\item 典范映射$\varprojlim\limits_{I}E_{\alpha}\to\varprojlim\limits_{J}E_{\alpha}$是双射.
		\item 若$I$有最大元$\omega$且$J=\left\{\omega\right\}$,则存在$\varprojlim\limits_{I}E_{\alpha}$到$E_{\omega}$的典范双射.
	\end{enumerate}
\end{proposition}

\begin{remark}{}{}
	\begin{enumerate}
		\item 对任意$\alpha\in I$,记$E_{\alpha}^{\prime}=f_{\alpha}\left(\varprojlim E_{\lambda}\right)$.
		\begin{itemize}
			\item $\left(E_{\alpha}^{\prime}\right)_{\alpha\in I}$是$\left(E_{\alpha}\right)_{\alpha\in I}$的子集逆系统,并且$\varprojlim E_{\alpha}^{\prime}=\varprojlim E_{\alpha}$.
			\item 对任意$\alpha,\beta\in I$且$\alpha\leq\beta$,记$f_{\alpha,\beta}^{\prime}:E_{\beta}^{\prime}\to E_{\alpha}^{\prime}$是$f_{\alpha,\beta}$的限制,则$\left(f_{\alpha,\beta}^{\prime}\right)_{\alpha,\beta\in I}$是一族满射.
			\item 对任意$\alpha\in I$有$E_{\alpha}^{\prime}\subseteq\bigcap\limits_{\substack{\beta\in I\\ \alpha\leq\beta}}f_{\alpha,\beta}\left(E_{\beta}\right)$.
		\end{itemize}
		\item 若$F$是集合,$I$是预序集,对任意$\alpha\in I$,$u_{\alpha}\in\Hom_{\mathsf{Set}}\left(F,E_{\alpha}\right)$满足$\forall\beta\in I\left(\alpha\leq\beta\implies f_{\alpha,\beta}\circ u_{\beta}=u_{\alpha}\right)$.考虑集合的逆向系统$\left(\left(F_{\alpha}\right)_{\alpha\in I},\left(i_{\alpha,\beta}\right)_{\alpha,\beta\in I}\right)$,其中对任意$\alpha\in I$有$F_{\alpha}=F$,对任意$\alpha,\beta\in I$且$\alpha\leq\beta$有$i_{\alpha,\beta}=\id_{F}$.
		\begin{itemize}
			\item $\varprojlim F_{\alpha}$典范等同于$F$.
			\item 对任意$\alpha\in I$,把$u_{\alpha}$视为$F_{\alpha}$到$E_{\alpha}$的映射,则$\left(u_{\alpha}\right)_{\alpha\in I}$是映射的逆向系统,其逆向极限等同于$\varprojlim u_{\alpha}$.
		\end{itemize}
		\item 设$I$是偏序集,对任意$J\in\fin\left(I\right)$记$F_{J}=\varprojlim\limits_{J}E_{\alpha}$,对任意$J,K\in\fin\left(I\right)$且$J\subseteq K$,$f_{JK}:F_{K}\to F_{J}$是典范映射.
		\begin{itemize}
			\item $\left(\left(F_{J}\right)_{J\in\fin\left(I\right)},\left(f_{JK}\right)_{J,K\in\fin\left(I\right)}\right)$是逆向极限.
			\item 设对任意$J\in\fin\left(I\right)$,$h_{J}:E\to F_{J}$是典范映射,则$\left(h_{J}\right)_{J\in\fin\left(I\right)}$是映射的逆向系统,$\varprojlim h_{J}$诱导典范双射
			\begin{align*}
				\varprojlim\limits_{I}E_{\alpha}\to\varprojlim\limits_{\fin\left(I\right)}\varprojlim\limits_{J}E_{\alpha}.
			\end{align*}
		\end{itemize}
	\end{enumerate}
\end{remark}
