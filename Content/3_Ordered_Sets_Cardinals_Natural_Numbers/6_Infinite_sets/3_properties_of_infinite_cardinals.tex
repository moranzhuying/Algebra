\section{无限基数的性质}

\begin{lemma}{}{}
	每一个无限集都有一个与$\N$等势的子集.
\end{lemma}

\begin{lemma}{}{}
	$\N\times\N$与$\N$等势.
\end{lemma}

\begin{theorem}{}{}
	对每个无限基数$\mathfrak{a}$都有$\mathfrak{a}^{2}=\mathfrak{a}$.
\end{theorem}

\begin{corollary}{}{}
	对每个无限基数$\mathfrak{a}$和每个$\geq 1$的自然数$n$都有$\mathfrak{a}^{n}=\mathfrak{a}$.
\end{corollary}

\begin{corollary}{}{}
	设$\left(\mathfrak{a}_{i}\right)_{i\in I}$是非零基数构成的有限族,则$\prod\limits_{i\in I}\mathfrak{a}_{i}=\max\limits_{i\in I}\mathfrak{a}_{i}$.
\end{corollary}

\begin{corollary}{}{}
	设$\mathfrak{a}$是无限基数,$\left(\mathfrak{a}_{i}\right)_{i\in I}$是一族$\leq\mathfrak{a}$的基数.若$\left|I\right|\leq\mathfrak{a}$,则
	\begin{enumerate}
		\item $\sum\limits_{i\in I}\mathfrak{a}_{i}=\mathfrak{a}$.
		\item 当$\exists j\in I\left(\mathfrak{a}_{j}=\mathfrak{a}\right)$时,$\sum\limits_{i\in I}\mathfrak{a}_{i}=\mathfrak{a}$.
	\end{enumerate}
\end{corollary}

\begin{corollary}{}{}
	设非零$\mathfrak{a},\mathfrak{b}$中至少有一个是无限基数,则$\mathfrak{a}\mathfrak{b}=\mathfrak{a}+\mathfrak{b}=\sup\left(\mathfrak{a},\mathfrak{b}\right)$.
\end{corollary}
