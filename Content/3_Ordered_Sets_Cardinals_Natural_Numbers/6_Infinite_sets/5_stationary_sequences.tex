\section{最终常值序列}

\begin{definition}{最终常值序列}{}
	设$\left(x_{n}\right)_{n=0}^{\infty}$是集合$E$中的序列.若$\exists m\in\N\forall n\in\N\left(n\geq m\implies x_{n}=x_{m}\right)$,则称$\left(x_{n}\right)_{n=0}^{\infty}$是最终常值序列.
\end{definition}

\begin{proposition}{偏序集的升链条件与极大元的存在性}{acc-equivalent-to-the-existence-of-maximal-elements}
	设$E$是偏序集,则$E$的每个非空子集都有极大元$\iff$ $E$的每个递增序列$\left(x_{n}\right)_{n=0}^{\infty}$都是最终常值序列.
\end{proposition}

\begin{definition}{Noether的偏序集}{}
	设$E$是偏序集.命题(\cref{prop:acc-equivalent-to-the-existence-of-maximal-elements})的条件之一成立时,称$E$是Noether的.
\end{definition}

\begin{corollary}{全序集的良序性与降链条件}{}
	设$E$是全序集,则$E$是良序集$\iff$ $E$的每个递减序列$\left(x_{n}\right)_{n=0}^{\infty}$都是最终常值序列.
\end{corollary}

\begin{corollary}{}{}
	有限偏序集中的每个递增序列都是最终常值序列.
\end{corollary}

\begin{proposition}{Noether归纳法原理}{}
	设$E$是Noether的偏序集,$F\subseteq E$.若$\forall a\in E\left(\forall x\in E\left(x>a\implies x\in F\right)\implies a\in F\right)$,则$F=E$.
\end{proposition}
