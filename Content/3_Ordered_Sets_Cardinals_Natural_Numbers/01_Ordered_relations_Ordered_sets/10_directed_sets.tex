\section{定向集}

\begin{definition}{预序集}{}
	设集合$E$配备预序$\leq$.
	\begin{enumerate}
		\item 若对任意$x,y\in E$有$\uparrow_{E}^{\leq}\left(\left\{x,y\right\}\right)\neq\emptyset$,则称$E$是(上)定向集.
		\item 若对任意$x,y\in E$有$\downarrow_{E}^{\leq}\left(\left\{x,y\right\}\right)\neq\emptyset$,则称$E$是下定向集.
	\end{enumerate}
\end{definition}

\begin{remark}{术语说明}{}
	当$E$是上定向集时,我们通常说``集合$E$按预序$\leq$向上定向''.
\end{remark}

\begin{example}{}{}
	\begin{enumerate}
		\item 如果$E$是具有最大元的偏序集,则$E$是上定向集.
		\item 如果$E$是拓扑空间,则每个$x\in E$的邻域基按包含偏序的对偶下定向.
		\item 如果$A$是环,则每个$A$-模的有限生成子模构成的集合按包含偏序上定向.
	\end{enumerate}
\end{example}

\begin{proposition}{}{}
	如果偏序集$E$是上定向的,则$E$的每个极大元都是$E$的最大元.
\end{proposition}

\begin{proposition}{}{}
	如果集合$E$配备预序$\leq$后是上定向集,$\leq^{\prime}$是其对偶,则$E$按$\leq^{\prime}$是下定向集.
\end{proposition}

\begin{proposition}{}{}
	如果$\left(E_{i}\right)_{i\in I}$是一族上定向集,则配备逐点序的$\prod\limits_{i\in I}E_{i}$是上定向集.
\end{proposition}

\begin{remark}{}{}
	如果$E$是上定向集,则$E$的子集不一定是上定向集,但$E$的每个共尾子集都是上定向集.
\end{remark}
