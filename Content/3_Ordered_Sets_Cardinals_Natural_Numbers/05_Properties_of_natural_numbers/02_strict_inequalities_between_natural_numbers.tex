\section{自然数之间的严格不等式}

\begin{proposition}{}{}
	设$a,b$是自然数,则$a<b$ $\iff$存在自然数$c$使得$c>0$且$b=a+c$.
\end{proposition}

\begin{proposition}{}{}
	设$\left(a_{i}\right)_{i\in I},\left(b_{i}\right)_{i\in I}$是自然数构成的有限族,并且$\forall i\in I\left(a_{i}\leq b_{i}\right)\wedge\exists j\in I\left(a_{j}<b_{j}\right)$.
	\begin{enumerate}
		\item $\sum\limits_{i\in I}a_{i}\leq\sum\limits_{i\in I}b_{i}$.
		\item 若$\forall i\in I\left(b_{i}>0\right)$,则$\prod\limits_{i\in I}a_{i}<\prod\limits_{i\in I}b_{i}$.
	\end{enumerate}
\end{proposition}

\begin{corollary}{}{}
	设$a,b,c$是自然数.若$a<b,c>0$,则$a^{c}<b^{c}$.
\end{corollary}

\begin{corollary}{}{}
	若自然数$a,b$满足$a\leq b$,则存在唯一的自然数$c$使得$b=a+c$.
\end{corollary}

\begin{definition}{自然数的差}{}
	设自然数$a,b$满足$a\leq b$.我们把满足$b=a+c$的自然数$c$记作$a-b$,称之为$a$和$b$的差.
\end{definition}

\begin{proposition}{}{}
	若$a,b,c,d$是自然数,$a\leq b,c\leq d$,则$\left(b-a\right)+\left(d-c\right)=\left(b+d\right)-\left(a+c\right)$.
\end{proposition}
