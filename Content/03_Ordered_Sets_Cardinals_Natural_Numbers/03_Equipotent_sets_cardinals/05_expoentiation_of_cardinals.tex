\section{基数的指数运算}

\begin{definition}{基数的指数}{}
	设$\mathfrak{a},\mathfrak{b}$是基数.我们把$\left|\Hom_{\mathsf{Set}}\left(\mathfrak{b},\mathfrak{a}\right)\right|$记作$\mathfrak{a}^{\mathfrak{b}}$.
\end{definition}

\begin{proposition}{}{}
	设$X,Y$是集合,则$\left|\Hom_{\mathsf{Set}}\left(X,Y\right)\right|=\left|Y\right|^{\left|X\right|}$.
\end{proposition}

\begin{proposition}{}{}
	设$\left(\mathfrak{a}_{i}\right)_{i\in I}$是一族基数,$\mathfrak{a},\mathfrak{b}$是基数,$\left|I\right|=\mathfrak{b}$.若$\forall i\in I\left(\mathfrak{a}_{i}=\mathfrak{a}\right)$,则$\mathfrak{a}^{\mathfrak{b}}=\prod\limits_{i\in I}\mathfrak{a}_{i}$.
\end{proposition}

\begin{corollary}{}{}
	设$\mathfrak{a}$是基数,$\left(\mathfrak{b}_{i}\right)_{i\in I}$是一族基数,$\mathfrak{b}=\sum\limits_{i\in I}\mathfrak{b}_{i}$,则$\mathfrak{a}^{\mathfrak{b}}=\prod\limits_{i\in I}\mathfrak{a}^{\mathfrak{b}_{i}}$.
\end{corollary}

\begin{corollary}{}{}
	设$\mathfrak{b}$是基数,$\left(\mathfrak{a}_{i}\right)_{i\in I}$是一族基数,则$\left(\prod\limits_{i\in I}\mathfrak{a}_{i}\right)^{\mathfrak{b}}=\prod\limits_{i\in I}\mathfrak{a}_{i}^{\mathfrak{b}}$.
\end{corollary}

\begin{corollary}{}{}
	对任意基数$\mathfrak{a},\mathfrak{b},\mathfrak{c}$有$\mathfrak{a}^{\mathfrak{b}\mathfrak{c}}=\left(\mathfrak{a}^{\mathfrak{b}}\right)^{\mathfrak{c}}$.
\end{corollary}

\begin{proposition}{}{}
	对任意基数$\mathfrak{a}$,有
	\begin{gather*}
		\mathfrak{a}^{0}=1,\mathfrak{a}^{1}=\mathfrak{a},1^{\mathfrak{a}}=1,\\
		\mathfrak{a}\neq 0\implies 0^{\mathfrak{a}}=0.
	\end{gather*}
\end{proposition}

\begin{proposition}{}{}
	设$X$是集合,则$\left|\mathcal{P}\left(X\right)\right|=2^{\left|X\right|}$.
\end{proposition}
