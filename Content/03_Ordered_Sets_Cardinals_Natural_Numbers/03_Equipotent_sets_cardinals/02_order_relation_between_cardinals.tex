\section{基数之间的序关系}

\begin{proposition}{}{}
	设$X,Y$是集合,则存在$X$到$Y$的单射$\iff$ $X$和$Y$的一个子集等势$\iff$ $\left|X\right|$和$\left|Y\right|$的一个子集等势.
\end{proposition}

\begin{theorem}{基数的良序性}{}
	把公式``$\mathfrak{r}$和$\mathfrak{n}$是基数,并且$\mathfrak{r}$和$\mathfrak{n}$的一个子集等势''记作$\mathfrak{r}\leq\mathfrak{n}$,则这是良序关系.
\end{theorem}

\begin{corollary}{基数的可比性}{}
	对任意两个集合$X$和$Y$,要么$\left|X\right|\leq\left|Y\right|$,要么$\left|Y\right|\leq\left|X\right|$.
\end{corollary}

\begin{corollary}{Cantor-Schroder-Bernstein定理}{}
	对任意集合$X$和$Y$,如果$\left|X\right|\leq\left|Y\right|$且$\left|Y\right|\leq\left|X\right|$,则$\left|X\right|=\left|Y\right|$.
\end{corollary}

\begin{remark}{}{}
	对任意基数$\mathfrak{a}$,所有满足$\mathfrak{r}\leq\mathfrak{a}$的基数$\mathfrak{r}$构成集合,称为$\leq\mathfrak{a}$的基数构成的集合.
\end{remark}

\begin{proposition}{基数的上确界的存在唯一性}{}
	对任意一族基数$\left(\mathfrak{a}_{i}\right)_{i\in I}$,存在唯一的基数$\mathfrak{b}$满足下列条件:
	\begin{itemize}
		\item $\forall i\in I\left(\mathfrak{a}_{i}\leq\mathfrak{b}\right)$.
		\item 对任意基数$\mathfrak{c}$有$\forall i\in I\left(\mathfrak{a}_{i}\leq\mathfrak{c}\implies\mathfrak{b}\leq\mathfrak{c}\right)$.
	\end{itemize}
	我们把$\mathfrak{b}$记作$\sup\limits_{i\in I}\mathfrak{a}_{i}$,称为$\left(\mathfrak{a}_{i}\right)_{i\in I}$的上确界.
\end{proposition}

\begin{proposition}{}{}
	设$X$和$Y$是集合.若存在从$X$到$Y$的基数,则$\left|Y\right|\leq\left|X\right|$.
\end{proposition}
