\section{基数\texorpdfstring{$0$}{0}和\texorpdfstring{$1$}{1}的性质}

\begin{proposition}{}{}
	设$\left(\mathfrak{a}_{i}\right)_{i\in I}$是一族基数,$J\subseteq I$.
	\begin{enumerate}
		\item 若对任意$i\in I\setminus J$有$\mathfrak{a}_{i}=0$,则$\sum\limits_{i\in I}\mathfrak{a}_{i}=\sum\limits_{i\in J}\mathfrak{a}_{i}$.
		\item 若对任意$i\in I\setminus J$有$\mathfrak{a}_{i}=1$,则$\prod\limits_{i\in I}\mathfrak{a}_{i}=\prod\limits_{i\in J}\mathfrak{a}_{i}$.
	\end{enumerate}
\end{proposition}

\begin{corollary}{}{}
	对任意基数$\mathfrak{a}$,有$\mathfrak{a}+0=\mathfrak{a},\mathfrak{a}1=\mathfrak{a}$.
\end{corollary}

\begin{corollary}{}{}
	设$\mathfrak{a}$和$\mathfrak{b}$是基数,$\left(\mathfrak{a}_{i}\right)_{i\in I},\left(\mathfrak{c}_{i}\right)_{i\in I}$是基数,$\left|I\right|=\mathfrak{a}$,对任意$i\in I$有$\mathfrak{a}_{i}=\mathfrak{a}\wedge\mathfrak{c}_{i}=1$,则$\mathfrak{a}\mathfrak{b}=\sum\limits_{i\in I}\mathfrak{a}_{i},\mathfrak{b}=\sum\limits_{i\in I}\mathfrak{b}_{i}$.
\end{corollary}

\begin{proposition}{}{}
	设$\left(\mathfrak{a}_{i}\right)_{i\in I}$是一族基数,则$\prod\limits_{i\in I}\mathfrak{a}_{i}\neq 0\iff \forall i\in I\left(\mathfrak{a}_{i}\neq 0\right)$.
\end{proposition}

\begin{proposition}{}{}
	若基数$\mathfrak{a}$和$\mathfrak{b}$满足$\mathfrak{a}+1=\mathfrak{b}+1$,则$\mathfrak{a}=\mathfrak{b}$.
\end{proposition}

\begin{remark}{}{}
	对任意基数$\mathfrak{a},\mathfrak{b},\mathfrak{c}$,$\left(\mathfrak{a}+\mathfrak{c}=\mathfrak{b}+\mathfrak{c}\implies\mathfrak{a}=\mathfrak{b}\right)$ $\iff$ $\mathfrak{m}$是有限基数.
\end{remark}
