\section{双重逆向极限}

\begin{proposition}{}{}
	设$I,J$是预序集,$\left(\left(E_{\alpha}^{\lambda}\right)_{\alpha\in I,\lambda\in J},\left(f_{\alpha\beta}^{\lambda\mu}\right)_{\alpha,\lambda\in I,\beta,\mu\in J}\right)$是集合的逆向系统,则存在典范双射$\varprojlim\limits_{I\times J}E_{\alpha}^{\lambda}\to\varprojlim\limits_{J}\varprojlim\limits_{I}E_{\alpha}^{\lambda}$.
\end{proposition}

\begin{corollary}{}{}
	设$I,J$是预序集,$\left(\left(E_{\alpha}^{\lambda}\right)_{\alpha\in I,\lambda\in J},\left(f_{\alpha\beta}^{\lambda\mu}\right)_{\alpha,\lambda\in I,\beta,\mu\in J}\right),\left(\left(F_{\alpha}^{\lambda}\right)_{\alpha\in I,\lambda\in J},\left(g_{\alpha\beta}^{\lambda\mu}\right)_{\alpha,\lambda\in I,\beta,\mu\in J}\right)$是集合的逆向系统,
	\begin{align*}
		\forall\alpha\in I\forall\lambda\in J\left(u_{\alpha}^{\lambda}\in\Hom_{\mathsf{Set}}\left(E_{\alpha}^{\lambda},F_{\alpha}^{\lambda}\right)\right).
	\end{align*}
	若$\left(u_{\alpha}^{\lambda}\right)_{\alpha\in I,\lambda\in J}$是映射的逆向系统,则$\varprojlim\limits_{I\times J}u_{\alpha}^{\lambda}=\varprojlim\limits_{J}\varprojlim\limits_{I}u_{\alpha}^{\lambda}$.
\end{corollary}

\begin{corollary}{}{}
	设$\left(\left(\left(E_{\alpha}^{\lambda}\right)_{\alpha\in I},\left(f_{\alpha\beta}^{\lambda\mu}\right)_{\alpha,\beta\in I}\right)\right)_{\lambda\in\Lambda}$是一族集合的逆向系统,则
	\begin{enumerate}
		\item $\left(\left(\prod\limits_{\lambda\in\Lambda}E_{\alpha}^{\lambda}\right)_{\alpha\in I},\left(\prod\limits_{\lambda\in\Lambda}f_{\alpha\beta}^{\lambda}\right)_{\alpha,\beta\in I}\right)$是集合的逆向系统.
		\item 存在典范双射$\varprojlim\limits_{I}\prod\limits_{\lambda\in\Lambda}E_{\alpha}^{\lambda}\to\prod\limits_{\lambda\in\Lambda}\varprojlim\limits_{I}E_{\alpha}^{\lambda}$.
	\end{enumerate}
\end{corollary}
