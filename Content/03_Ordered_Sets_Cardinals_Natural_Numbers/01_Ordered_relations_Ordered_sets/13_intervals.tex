\section{区间}

\begin{definition}{有界区间}{}
	设集合$E$配备偏序$\leq$,$a,b\in E$且$a\leq b$.定义
	\begin{gather*}
		\left[a,b\right]\coloneq\left\{x\in E\middle|a\leq x\leq b\right\},\left[a,b\right)\coloneq\left\{x\in E\middle|a\leq x<b\right\},\\
		\left(a,b\right]\coloneq\left\{x\in E\middle|a<x\leq b\right\},\left(a,b\right)\coloneq\left\{x\in E\middle|a<x<b\right\}.
	\end{gather*}
	\begin{itemize}
		\item $\left[a,b\right]$称为以$a$为左端点$b$为右端点的闭区间.
		\item $\left[a,b\right)$称为以$a$为左端点$b$为右端点的左闭右开区间.
		\item $\left(a,b\right]$称为以$a$为左端点$b$为右端点的左开右闭区间区间.
		\item $\left(a,b\right)$称为以$a$为左端点$b$为右端点的开区间.
	\end{itemize}
\end{definition}

\begin{definition}{无界区间}{}
	设集合$E$配备偏序$\leq$,$a\in E$.定义
	\begin{gather*}
		\left(-\infty,a\right]\coloneq\left\{x\in E\middle|x\leq a\right\},\left(-\infty,a\right)\coloneq\left\{x\in E\middle|x<a\right\},\\
		\left[a,+\infty\right)\coloneq\left\{x\in E\middle|x\geq a\right\},\left(a,+\infty\right)\coloneq\left\{x\in E\middle|x>a\right\},\\
		\left(-\infty,+\infty\right)\coloneq E.
	\end{gather*}
	\begin{itemize}
		\item $\left(-\infty,a\right]$称为以$a$为右端点的左闭无界区间,$\left(-\infty,a\right)$称为以$a$为右端点的左开无界区间.
		\item $\left[a,+\infty\right)$称为以$a$为左端点的右闭无界区间,$\left(-\infty,a\right)$称为以$a$为左端点的右开无界区间.
	\end{itemize}
\end{definition}

\begin{proposition}{}{}
	设集合$E$配备偏序$\leq$后是格,则$E$中任意两个区间的交还是$E$中的区间.
\end{proposition}
