\section{Zermelo定理}

\begin{lemma}{}{}
	设$E$是集合,$\mathfrak{S}\subseteq\mathcal{P}\left(E\right)$,$p:\mathfrak{S}\to E$满足$\forall X\in\mathfrak{S}\left(p\left(X\right)\notin X\right)$,则存在$M\subseteq E$和$M$上的良序$\leq$,使得
	\begin{itemize}
		\item 对任意$x\in M$有$S_{x}\in\mathfrak{S}$且$p\left(S_{x}\right)=x$;
		\item $M\notin\mathfrak{S}$.
	\end{itemize}
\end{lemma}

\begin{theorem}{Zermelo定理(良序定理)}{}
	每个集合$E$上都存在一个良序.
\end{theorem}
