\section{良序集之间的同构}

\begin{lemma}{}{}
	设$E,F$是良序集,$f:E\to F$是序同构,$\im\left(f\right)$是$F$的始段,$g:E\to F$严格递增,则在逐点序下有$f\leq g$.
\end{lemma}

\begin{theorem}{良序集的比较定理}{}
	设$E$和$F$都是良序集,则下列陈述有且仅有一个成立:
	\begin{itemize}
		\item 存在从$E$到$F$的一个始段的序同构.
		\item 存在从$F$到$E$的一个始段的序同构.
	\end{itemize}
\end{theorem}

\begin{corollary}{良序集的刚醒}{}
	设$E$是良序集,则从$E$到$E$的一个始段的序同构只可能有$\id_{E}$.
\end{corollary}
