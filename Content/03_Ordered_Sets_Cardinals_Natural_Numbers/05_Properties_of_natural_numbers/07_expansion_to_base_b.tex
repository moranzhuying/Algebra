\section{\texorpdfstring{$b$}{b}进展开}

\begin{proposition}{}{}
	设$b$是$\geq 1$的自然数,对每个$>0$的自然数$k$,$E_{k}$是$\left(\left[0,k-1\right]\right)_{i=0}^{k-1}$的字典序乘积,定义
	\begin{align*}
		f_{k}:E_{k}\to\left[0,b^{k}-1\right],\left(r_{h}\right)_{i=0}^{k-1}\mapsto\sum\limits_{h=0}^{k-1}r_{h}b^{k-h-1},
	\end{align*}
	则对每个$>0$的自然数$k$,映射$f_{k}$是序同构.
\end{proposition}

\begin{proposition}{}{}
	设$b$是$\geq 1$的自然数,则对每个自然数$a$都有$a<b^{a}$.
\end{proposition}

\begin{proposition}{}{}
	对每个自然数$a$,存在唯一的自然数$k$和$\left[0,b-1\right]$中唯一的自然数序列$\left(r_{h}\right)_{h=0}^{k-1}$使得$r_{0}>0\wedge a=\sum\limits_{h=0}^{k-1}r_{h}b^{k-h-1}$.
\end{proposition}

\begin{definition}{$b$进展开式,计数符号}{}
	设$a,k$是自然数,$\left(r_{h}\right)_{h=0}^{k-1}$是$\left[0,b-1\right]$中满足$r_{0}>0\wedge a=\sum\limits_{h=0}^{k-1}r_{h}b^{k-h-1}$的序列.
	\begin{enumerate}
		\item 我们称$\sum\limits_{h=0}^{k-1}r_{h}b^{k-h-1}$是$a$的$b$进制展开.
		\item $\left(r_{h}\right)_{h=0}^{k-1}$的每一项称为一个数字.
		\item 我们称表达式$\left(\overline{r_{0}\ldots r_{k-2}r_{k-1}}\right)_{b}$是$a$的$b$进制表达式.
	\end{enumerate}
\end{definition}
