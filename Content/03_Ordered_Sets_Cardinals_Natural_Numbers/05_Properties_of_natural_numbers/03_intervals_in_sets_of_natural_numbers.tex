\section{自然数集中的区间}

\begin{definition}{自然数集中的闭区间}{}
	设$n$是自然数.我们把所有满足$a\leq n$的基数$a$组成的集合记作$\left[0,a\right]$.
\end{definition}

\begin{proposition}{}{}
	设$a,b$是自然数,则映射$\left[0,b\right]\to\left[a,a+b\right],x\mapsto a+x$是序同构,其逆为$\left[a,a+b\right]\to\left[0,b\right],x\mapsto x-a$.
\end{proposition}

\begin{proposition}{}{}
	设$a,b$是自然数且$a\leq b$,则$\left[a,b\right]$是有限集,其基数为$b-a+1$.
\end{proposition}

\begin{proposition}{}{}
	对每个$\geq 1$的自然数$n$和每个基数为$n$且配备了全序的偏序集,存在唯一的序同构$E\to\left[1,n\right]$.
\end{proposition}
