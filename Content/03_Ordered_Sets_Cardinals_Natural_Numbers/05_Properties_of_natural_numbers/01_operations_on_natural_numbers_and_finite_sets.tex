\chapter{自然数的运算,有限集}

\section{整数和有限集的运算}

\begin{proposition}{}{}
	设$\left(a_{i}\right)_{i\in I}$是自然数构成的有限族,则$\sum\limits_{i\in I}a_{i}$和$\prod\limits_{i\in I}a_{i}$都是自然数.
\end{proposition}

\begin{corollary}{}{}
	由有限集构成的有限族的并还是有限集.
\end{corollary}

\begin{corollary}{}{}
	由有限集构成的有限族的乘积还是有限集.
\end{corollary}

\begin{corollary}{}{}
	若$a,b$是自然数,则$a^{b}$是自然数.
\end{corollary}

\begin{corollary}{}{}
	有限集的幂集是有限集.
\end{corollary}
