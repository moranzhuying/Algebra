\chapter{主理想整环}

\section{主理想整环的定义}

\begin{definition}{主理想整环}{}
	设$A$是整环.若$A$的每个理想都是主理想,则称$A$是主理想整环(或PID).
\end{definition}

\begin{example}{PID的例子}{}
	\begin{enumerate}
		\item $\Z$是PID.
		\item 设$K$是域,则$K[X],K[[X]]$是PID.
		\item $\mathfrak{p}$-进域中的整数环是PID.
		\item 设$\mathrm{i}$是$X^{2}+1\in\Q[X]$的根,则$\Z[i]\coloneq\left\{a+b\mathrm{i}\in\Q(\mathrm{i})\middle|a,b\in\Z\right\}$是$\Q(\mathrm{i})$的子环,称为Gauss整数环,这也是PID.
		\item 设$\rho$是$X^{2}+5\in\Q[X]$的根,则$\Z[\rho]\coloneq\left\{a+b\rho\in\Q(\rho)\middle|a,b\in\Z\right\}$是$\Q(\rho)$的子环,但不是PID.
		\item 设$K$是域,则$K[X,Y]$不是PID.
	\end{enumerate}
\end{example}
