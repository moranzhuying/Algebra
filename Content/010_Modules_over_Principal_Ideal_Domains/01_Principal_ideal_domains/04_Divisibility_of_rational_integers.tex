\section{整数环上的整除理论}

\begin{proposition}{$\Z$的不可约元代表系}{}
	\begin{enumerate}
		\item 乘法群$\Q_{>0}$在$\Q$中的每个相伴类都是单点集,并且$\Q_{>0}$同构于$\mathscr{P}^{\ast}\left(\Q\right)$.
		\item $\Z$中每个$>0$的不可约元$x$都是素数(也称为有理素数),并且一定存在$p\in\mathbb{P}$使得$x\in\left\{-p,p\right\}$.
	\end{enumerate}
\end{proposition}

\begin{remark}{}{}
	基于典范同构$\Q_{>0}\simeq\mathscr{P}^{\ast}\left(\Q\right)$,讨论$\Q$中的元素族(相对于$\Z$)的最大公因子或最小公倍数时,通常默认取正的代表元.如此一来,任意一族有理数的最大公因数和最小公倍数是唯一的.
\end{remark}

\begin{proposition}{素数的无限性}{}
	素数集$\mathbb{P}$是无限集.
\end{proposition}
