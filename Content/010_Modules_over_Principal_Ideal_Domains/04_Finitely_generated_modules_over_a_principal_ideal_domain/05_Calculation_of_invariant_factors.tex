\section{不变因子的计算}

记号约定:
\begin{itemize}
	\item $A$是PID,$M$是自由$A$-模,$\mathcal{B}\coloneq\left(e_{i}\right)_{i=1}^{m}$是$M$的基.
	\item $N$是$M$的子模,$\left(z_{j}\right)_{j=1}^{n}$是$N$的生成系.
	\item $\left(c_{i,j}\right)_{\substack{1\leq i\leq m\\ 1\leq j\leq n}}\in\mathbf{M}_{m,n}\left(A\right)$满足下列条件:对每个$1\leq j\leq n$有$z_{j}=\sum\limits_{i=1}^{m}e_{i}c_{i,j}$.
	\item $\left(A\alpha_{i}\right)_{i=1}^{n}$是$N$相对$M$的不变因子序列.
	\item 对每个$1\leq j\leq n$,记$\delta_{j}\coloneq\alpha_{1}\ldots\alpha_{j}$.
\end{itemize}

\begin{proposition}{行列式因子与不变因子的关系}{}
	对每个$1\leq j\leq n$,$\delta_{m}$是$\left(c_{i,j}\right)_{\substack{1\leq i\leq m\\ 1\leq j\leq n}}$的$j$阶子式的一个最大公因子.
\end{proposition}
